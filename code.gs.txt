function createSheetIfNeeded(name, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const existing = ss.getSheetByName(name);
  if (!existing) {
    const s = ss.insertSheet(name);
    s.getRange(1, 1, 1, headers.length).setValues([headers]);
    s.setFrozenRows(1);
    return;
  }
  // ── Migrasi: tambahkan kolom baru yang belum ada di header existing ──
  const lastCol = existing.getLastColumn();
  const currentHeaders = lastCol > 0
    ? existing.getRange(1, 1, 1, lastCol).getValues()[0]
    : [];
  const missing = headers.filter(h => currentHeaders.indexOf(h) === -1);
  if (missing.length > 0) {
    existing.getRange(1, currentHeaders.length + 1, 1, missing.length).setValues([missing]);
    Logger.log(`[MIGRATION] Sheet "${name}" — kolom ditambahkan: ${missing.join(', ')}`);
  }
}

function checkForcePasswordChange(username) {
  try {
    if (String(username).toLowerCase() !== 'admin') return { forced: false };
    const flag = getCfg('force_password_change_admin');
    return { forced: String(flag).toLowerCase() === 'true' };
  } catch(e) { return { forced: false }; }
}

/**
 * Validasi password policy korporasi
 * - Minimal 8 karakter
 * - Minimal 1 huruf besar
 * - Minimal 1 huruf kecil  
 * - Minimal 1 angka
 * - Minimal 1 simbol
 * - Tidak boleh mengandung username
 * - Tidak boleh password umum (top 100)
 */
function validatePasswordPolicy(password, username) {
  if (!password || typeof password !== 'string') {
    return { valid: false, reason: 'Password tidak valid' };
  }
  
  const COMMON_PASSWORDS = [
    'password', '123456', 'qwerty', 'admin', 'letmein', 'welcome',
    'monkey', 'dragon', 'master', 'login', 'abc123', 'iloveyou'
  ];
  
  if (password.length < 8) return { valid: false, reason: 'Minimal 8 karakter' };
  if (password.length > 128) return { valid: false, reason: 'Maksimal 128 karakter' };
  if (!/[A-Z]/.test(password)) return { valid: false, reason: 'Harus mengandung huruf besar' };
  if (!/[a-z]/.test(password)) return { valid: false, reason: 'Harus mengandung huruf kecil' };
  if (!/[0-9]/.test(password)) return { valid: false, reason: 'Harus mengandung angka' };
  if (!/[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password)) {
    return { valid: false, reason: 'Harus mengandung simbol' };
  }
  if (username && password.toLowerCase().includes(username.toLowerCase())) {
    return { valid: false, reason: 'Password tidak boleh mengandung username' };
  }
  if (COMMON_PASSWORDS.includes(password.toLowerCase())) {
    return { valid: false, reason: 'Password terlalu umum' };
  }
  
  return { valid: true };
}

/**
 * Fungsi untuk user mengganti password sendiri (panggil dari Settings)
 */
function changePassword(oldPassword, newPassword, confirmPassword, username, token) {
  try {
    if (!username) return { success: false, error: 'Username wajib diisi' };
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };
    
    // Gunakan ctx.username alih-alih Session.getActiveUser().getEmail()
    const targetUser = ctx.username; 
    
    if (newPassword !== confirmPassword) {
      return { success: false, error: 'Konfirmasi password tidak cocok' };
    }
    
    const policy = validatePasswordPolicy(newPassword, username.split('@')[0]);
    if (!policy.valid) {
      return { success: false, error: policy.reason };
    }
    
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.MASTER_USER);
    const h = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
    
    const ui = h.indexOf('username');
    const pi = h.indexOf('passwordHash');
    
    let userRowIdx = -1;
    for (let i = 0; i < rows.length; i++) {
      if (String(rows[i][ui]).toLowerCase() === username.toLowerCase()) {
        userRowIdx = i;
        break;
      }
    }
    
    if (userRowIdx === -1) return { success: false, error: 'User tidak ditemukan' };
    
    // Verifikasi password lama
    if (!verifyPassword(oldPassword, String(rows[userRowIdx][pi] || ''))) {
      logFailedLoginAttempt(username);
      return { success: false, error: 'Password lama salah' };
    }
    
    // Update ke hash baru
    rows[userRowIdx][pi] = hashPassword(newPassword);
    rows[userRowIdx][h.indexOf('updatedAt')] = new Date();
    
    s.getRange(userRowIdx + 2, 1, 1, h.length).setValues([rows[userRowIdx]]);
    logAudit(username, 'PASSWORD_CHANGE', 'Auth', null);

    if (username.toLowerCase() === 'admin') {
      const flagRow = getMasterData(CONFIG.SHEETS.MASTER_CONFIG)
                        .find(c => c.configKey === 'force_password_change_admin');
      if (flagRow) saveMasterData(CONFIG.SHEETS.MASTER_CONFIG, { configValue: 'false' }, username, flagRow.rowId);
    }
    
    return { success: true, message: 'Password berhasil diganti' };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

// ==================== AUTHORIZATION HELPERS ====================
function getCurrentUserContext(username, token) {
  if (!username) return null;
  
  const sp = PropertiesService.getScriptProperties();
  const sessionKey = 'SESSION_' + String(username).toUpperCase();
  const raw = sp.getProperty(sessionKey);
  
  if (!raw) return null;
  
  try {
    const sess = JSON.parse(raw);
    if (Date.now() > (sess.expiresAt || 0)) {
      sp.deleteProperty(sessionKey);
      return null;
    }
    // ✅ INI PERBAIKANNYA — token wajib cocok dengan yang tersimpan di sesi
    if (!token || sess.token !== token) {
      logAudit(username, 'SESSION_TOKEN_MISMATCH', 'Auth', null);
      return null;
    }
    return {
      username: sess.username,
      fullName: sess.fullName,
      unitCode: String(sess.unitCode || 'ALL').toUpperCase(),
      isAdmin: String(sess.unitCode || '').toUpperCase() === 'ALL'
    };
  } catch (e) {
    return null;
  }
}

function canAccessFlight(username, token, flightRecord) {
  const ctx = getCurrentUserContext(username, token);
  if (!ctx) return { allowed: false, reason: 'Session tidak valid' };
  if (ctx.isAdmin) return { allowed: true };
  if (!flightRecord) return { allowed: false, reason: 'Data tidak ditemukan' };
  
  const flightUnit = String(flightRecord.unitCode || '').toUpperCase();
  if (flightUnit !== ctx.unitCode) {
    logAudit(username, 'UNAUTHORIZED_ACCESS_ATTEMPT', 'Flights', 
             flightRecord.rowId, null, null, null, null, 
             `Tried to access ${flightUnit} data as ${ctx.unitCode}`);
    return { allowed: false, reason: 'Tidak memiliki akses ke unit ini' };
  }
  return { allowed: true };
}

/**
 * Check apakah user boleh melakukan aksi admin-only
 */
function requireAdmin(username) {
  const ctx = getCurrentUserContext(username);
  if (!ctx) throw new Error('Session tidak valid');
  if (!ctx.isAdmin) {
    logAudit(username, 'UNAUTHORIZED_ADMIN_ATTEMPT', 'System', null);
    throw new Error('Akses ditolak: diperlukan hak administrator');
  }
  return ctx;
}

/**
 * Extract username dari parameter pertama (konvensi untuk semua public function)
 */
function extractUsername(firstParam) {
  // Jika parameter pertama adalah object dengan field _username
  if (firstParam && typeof firstParam === 'object' && firstParam._username) {
    return firstParam._username;
  }
  // Fallback ke Session (tidak reliable tapi sebagai backup)
  return Session.getActiveUser().getEmail() || null;
}

// ==================== UTILITY ====================
function fmtDate(v) {
  if (!v) return "";
  if (v instanceof Date) {
    return v.getFullYear() + '-' +
           String(v.getMonth() + 1).padStart(2,'0') + '-' +
           String(v.getDate()).padStart(2,'0');
  }
  return String(v).split("T")[0];
}

function fmtTime(v) {
  if (!v) return "";
  if (v instanceof Date) {
    return String(v.getHours()).padStart(2,'0') + ':' +
           String(v.getMinutes()).padStart(2,'0');
  }
  const m = String(v).match(/(\d{2}):(\d{2})/);
  return m ? `${m[1]}:${m[2]}` : String(v);
}

function toMin(t) {
  if (!t) return 0;
  const p = String(t).split(':');
  return parseInt(p[0] || 0) * 60 + parseInt(p[1] || 0);
}

function sv(v) {
  if (v instanceof Date) return v.toISOString();
  return (v === null || v === undefined) ? '' : v;
}

/**
 * ✅ FIX: Sanitasi REKURSIF untuk seluruh pohon objek/array sebelum
 * di-return lewat google.script.run pada endpoint PUBLIK (tanpa sesi).
 * Objek Date yang bersarang di dalam array/objek (bukan di level teratas)
 * bisa membuat SELURUH payload RPC gagal diserialisasi dan client
 * menerima null utuh — bukan error per-field seperti JSON.stringify biasa.
 * sv() saja tidak cukup karena hanya menyanitasi satu nilai, bukan
 * struktur bersarang seperti { data: [ { updatedAt: Date }, ... ] }.
 */
function _deepSanitizeForClient(obj) {
  if (obj === null || obj === undefined) return null;
  if (obj instanceof Date) return obj.toISOString();
  if (Array.isArray(obj)) return obj.map(_deepSanitizeForClient);
  if (typeof obj === 'object') {
    const out = {};
    Object.keys(obj).forEach(function(k) {
      out[k] = _deepSanitizeForClient(obj[k]);
    });
    return out;
  }
  return obj; // string, number, boolean sudah aman dikirim apa adanya
}

// ==================== EXCHANGE RATE PER TANGGAL ====================
/**
 * Ambil kurs berdasarkan currency + tanggal flight.
 * Mengembalikan nilai kurs (number) atau null jika belum ada.
 * Dipanggil dari frontend SEBELUM modal validasi terbuka.
 */
function getExchangeRateByDate(currency, flightDate) {
  try {
    if (!currency || !flightDate) return null;
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.MASTER_EXCHANGE_RATE);
    if (!s || s.getLastRow() < 2) return null;

    const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();

    const ci = h.indexOf('currency');
    const ri = h.indexOf('rate');
    const ui = h.indexOf('updatedAt');
    if (ci === -1 || ri === -1 || ui === -1) return null;

    // Normalkan tanggal target ke midnight
    const target = new Date(flightDate);
    target.setHours(0, 0, 0, 0);

    for (let i = 0; i < rows.length; i++) {
      const rowCur = String(rows[i][ci]).toUpperCase().trim();
      if (rowCur !== currency.toUpperCase()) continue;

      const rawDate = rows[i][ui];
      if (!rawDate) continue;
      const rowDate = rawDate instanceof Date ? new Date(rawDate) : new Date(rawDate);
      rowDate.setHours(0, 0, 0, 0);

      if (rowDate.getTime() === target.getTime()) {
        const rate = parseFloat(rows[i][ri]);
        return isNaN(rate) ? null : rate;
      }
    }
    return null; // belum ada kurs untuk tanggal ini

  } catch(e) {
    Logger.log('getExchangeRateByDate error: ' + e.toString());
    return null;
  }
}

function saveExchangeRateByDate(currency, rate, flightDate, uid, token) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    if (!currency || !rate || !flightDate) {
      return { success: false, error: 'Parameter tidak lengkap' };
    }
    
    // ── AUTHORIZATION: hanya admin yang boleh simpan kurs manual ──
    const ctx = getCurrentUserContext(uid, token);
    if (!ctx) return { success: false, error: 'Session expired' };
    if (!ctx.isAdmin) {
      logAudit(uid, 'UNAUTHORIZED_EXCHANGE_RATE_EDIT', CONFIG.SHEETS.MASTER_EXCHANGE_RATE, null);
      return { success: false, error: 'Hanya admin yang dapat mengubah kurs' };
    }
    
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.MASTER_EXCHANGE_RATE);
    if (!s) return { success: false, error: 'Sheet tidak ditemukan' };

    const h = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const ci  = h.indexOf('currency');
    const ri  = h.indexOf('rate');
    const si  = h.indexOf('source');
    const ui  = h.indexOf('updatedAt');
    const ubi = h.indexOf('updatedBy');
    
    if (ci === -1 || ri === -1 || ui === -1) {
      return { success: false, error: 'Kolom Master_ExchangeRate tidak lengkap' };
    }

    // Normalkan tanggal target ke midnight
    const target = new Date(flightDate);
    target.setHours(0, 0, 0, 0);

    // Cari baris yang sudah ada
    let existingRowNum = -1;
    if (s.getLastRow() >= 2) {
      const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
      for (let i = 0; i < rows.length; i++) {
        const rowCur = String(rows[i][ci]).toUpperCase().trim();
        if (rowCur !== currency.toUpperCase()) continue;
        const rawDate = rows[i][ui];
        if (!rawDate) continue;
        const rowDate = rawDate instanceof Date ? new Date(rawDate) : new Date(rawDate);
        rowDate.setHours(0, 0, 0, 0);
        if (rowDate.getTime() === target.getTime()) {
          existingRowNum = i + 2; // +2: row 1 = header
          break;
        }
      }
    }

    const rateVal = parseFloat(rate);
    const actorUsername = ctx.username; // ✅ Ambil langsung dari ctx yang sudah tervalidasi

    if (existingRowNum > 0) {
      // UPDATE baris yang sudah ada
      const rowData = s.getRange(existingRowNum, 1, 1, h.length).getValues()[0];
      rowData[ri] = rateVal;
      rowData[ui] = target;
      if (si  !== -1) rowData[si]  = 'Input validasi';
      if (ubi !== -1) rowData[ubi] = actorUsername; // ✅ Lebih bersih dan aman
      
      s.getRange(existingRowNum, 1, 1, h.length).setValues([rowData]);
      logAudit(actorUsername, 'UPDATE', CONFIG.SHEETS.MASTER_EXCHANGE_RATE, existingRowNum);
      return { success: true, action: 'updated', rowId: existingRowNum };

    } else {
      // INSERT baris baru
      const newRow = h.map(col => {

function updateFlight(rid, updates, uid, auditAction = 'UPDATE') {
  const s = SpreadsheetApp.getActiveSpreadsheet()
              .getSheetByName(CONFIG.SHEETS.FLIGHTS);
  const h   = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
  const row = s.getRange(rid, 1, 1, s.getLastColumn()).getValues()[0];
  const old = [...row];
  const IMMUTABLE_AFTER_INVOICE = [
    'grossAmount', 'vatAmount', 'pphAmount', 'totalAmount',
    'rateUsed', 'kursUsed', 'invoiceNo',
    'vatPctSnapshot', 'pphPctSnapshot', 'validatedAt', 'validatedBy',
    'signatorySnapshot'
  ];
  const sfIdx = h.indexOf('statusFlow');
  const currentStatus = sfIdx !== -1 ? row[sfIdx] : '';
  const isLocked = ['INVOICED','UNPAID','PAID'].includes(String(currentStatus));

  if (isLocked) {
    IMMUTABLE_AFTER_INVOICE.forEach(field => {
      if (field in updates) {
        // Hanya boleh diubah oleh proses validasi itu sendiri (updates.validatedAt ada)
        if (!updates.validatedAt) {
          delete updates[field];
          Logger.log(`[GUARD] Field '${field}' dilindungi — tidak dapat diubah setelah ${currentStatus}`);
        }
      }
    });
  }

  Object.keys(updates).forEach(k => {
    const i = h.indexOf(k);
    if (i !== -1) row[i] = updates[k];
  });
  const ua = h.indexOf("updatedAt"), ub = h.indexOf("updatedBy");
  if (ua !== -1) row[ua] = new Date();
  if (ub !== -1) row[ub] = (uid && String(uid).trim()) ? String(uid).trim()
  : (Session.getActiveUser().getEmail() || "sys");
  s.getRange(rid, 1, 1, h.length).setValues([row]);
  SpreadsheetApp.flush(); // ✅ FIX: paksa commit sebelum function return —
                          // mencegah fetch berikutnya (dari device yang sama)
                          // membaca data STALE yang membuat baris "hidup lagi"
  if (auditAction) logAudit(uid, auditAction, "Flights", rid); // false = tanpa log (cegah notif ganda)
  return { success: true };
}

function softDeleteFlight(rid, uid) {
  const f = getFlightByRowId ? getFlightByRowId(rid) : getAllFlights().find(x => x.rowId == rid);
  if (!f) return { success: false, error: 'Data tidak ditemukan' };
  if (['INVOICED','UNPAID','PAID'].includes(String(f.statusFlow))) {
    return { success: false, error: 'Data yang sudah memiliki invoice tidak dapat dihapus. Gunakan fitur Void/Batalkan Invoice.' };
  }
  // ✅ BARU TAHAP 1: blokir hapus jika flight sedang terikat batch aktif
  if (f.batchId && String(f.batchId).trim()) {
    const batch = _getBatchById(String(f.batchId).trim());
    if (batch && ['SENT','PROOF_SUBMITTED'].includes(batch.status)) {
      return { success: false, error: 'Flight ini sedang terikat batch invoice aktif. Selesaikan atau tolak batch terlebih dahulu.' };
    }
  }
  const result = updateFlight(rid, { isDeleted: true }, uid, false);
  if (result.success) {
    logAudit(uid, 'DELETE', 'Flights', rid, null, null, null, null,
      (f.acid || '') + ' — ' + (f.flightDate || ''));
  }
  return result;
}

function voidInvoice(rid, uid, reason) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const f = getFlightByRowId ? getFlightByRowId(rid) : getAllFlights().find(x => x.rowId == rid);
    if (!f) return { success: false, error: 'Data tidak ditemukan' };
    if (!['INVOICED','UNPAID'].includes(String(f.statusFlow))) {
      return { success: false, error: 'Hanya invoice berstatus INVOICED/UNPAID yang bisa dibatalkan' };
    }
    // ✅ BARU TAHAP 1: blokir void jika flight sedang terikat batch aktif
    if (f.batchId && String(f.batchId).trim()) {
      const batch = _getBatchById(String(f.batchId).trim());
      if (batch && ['SENT','PROOF_SUBMITTED'].includes(batch.status)) {
        return { success: false, error: 'Flight ini sedang terikat batch invoice aktif. Selesaikan atau tolak batch terlebih dahulu.' };
      }
    }
    // Soft-void: ubah status ke VOID & sembunyikan dari daftar aktif (isDeleted=true).
    // Sistem tidak menyimpan file PDF di Drive (dokumen dibangun on-demand di
    // client saat Preview/Download), sehingga tidak ada file yang dipindahkan
    // ke Trash. Data & riwayat tetap ada di sheet + Audit Log.
    updateFlight(rid, {
      statusFlow: 'VOID',
      isDeleted: true,
      validatedAt: f.validatedAt 
    }, uid, false);
    logAudit(uid, 'VOID', "Flights", rid);
    return { success: true };
  } finally {
    lock.releaseLock();
  }
}

// ==================== MASTER DATA ====================
function getMasterData(sheet, incDel = false) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const s  = ss.getSheetByName(sheet);
  if (!s || s.getLastRow() < 2) return [];

  const h        = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
  const d        = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
  const di       = h.indexOf("isDeleted");
  const timeZone = ss.getSpreadsheetTimeZone();

  return d.reduce((a, row, i) => {
    if (!incDel && di !== -1 &&
        (row[di] === true || row[di] === "TRUE")) return a;
    const o = { rowId: i + 2 };
    h.forEach((k, j) => {
      let val = row[j];
      // Format waktu khusus untuk Master_AirportHours
      if (sheet === 'Master_AirportHours' &&
          (k === 'normalStart' || k === 'normalEnd')) {
        if (val instanceof Date && !isNaN(val.getTime())) {
          val = Utilities.formatDate(val, timeZone, "HH:mm");
        } else if (typeof val === 'string' && val.includes(':')) {
          const m = val.match(/(\d{1,2}):(\d{2})/);
          val = m ? m[1].padStart(2,'0') + ':' + m[2] : val;
        }
      }
      o[k] = val;
    });
    a.push(o);
    return a;
  }, []);
}

        const uaIdx = h.indexOf("updatedAt");
        const ubIdx = h.indexOf("updatedBy");
        if (uaIdx !== -1) existingRow[uaIdx] = new Date();

        s.getRange(rid, 1, 1, h.length).setValues([existingRow]);

           .find(a => a.airportCode === code && a.isActive !== false) || null;
}

function calcDuration(timeStr, unitCode) {
  const rule = getAirportRule(unitCode);
  if (!rule) return { category: "NORMAL", actualHours: 0, billableHours: 0, isNormal: true };

  const t      = toMin(fmtTime(timeStr));
  const buka   = toMin(fmtTime(rule.normalStart));
  const tutup  = toMin(fmtTime(rule.normalEnd));
  const minDur = rule.minDuration ? parseInt(rule.minDuration) : 1;

  if (t < buka) {
    const diff          = buka - t;
    const actualHours   = diff / 60;
    // Selalu bulatkan ke atas per jam penuh
    let   billableHours = Math.ceil(diff / 60);
    if (billableHours < minDur) billableHours = minDur;
    return { category: "ADVANCE", actualHours, billableHours, duration: billableHours, isNormal: false };
  }
  if (t > tutup) {
    const diff          = t - tutup;
    const actualHours   = diff / 60;
    // Selalu bulatkan ke atas per jam penuh
    let   billableHours = Math.ceil(diff / 60);
    if (billableHours < minDur) billableHours = minDur;
    return { category: "EXTEND", actualHours, billableHours, duration: billableHours, isNormal: false };
  }
  return { category: "NORMAL", actualHours: 0, billableHours: 0, isNormal: true };
}

// Fungsi hitungDurasi yang BENAR — memanggil calcDuration
function hitungDurasi(unitCode, trafficType, waktuWita, category) {
  try {
    if (!unitCode || !waktuWita || !category) {
      return { success: false, error: 'Parameter tidak lengkap', duration: 0 };
    }
    
    const result = calcDuration(waktuWita, unitCode);
    
    // Jika kategori tidak cocok dengan hasil kalkulasi, gunakan manual
    if (result.category !== category) {
      return { 
        success: true, 
        duration: 0,
        category: category,
        warning: `Kategori ${category} tidak sesuai dengan waktu operasi bandara`
      };
    }
    
    return { 
      success: true, 
      duration: result.billableHours || result.duration || 0,
      category: result.category,
      actualHours: result.actualHours
    };
  } catch(e) {
    return { success: false, error: e.toString(), duration: 0 };
  }
}

function genInvoiceNo(uc, di, uid) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const s  = SpreadsheetApp.getActiveSpreadsheet()
                 .getSheetByName(CONFIG.SHEETS.MASTER_SEQUENCE);
    const h  = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getLastRow() > 1
      ? s.getRange(2, 1, s.getLastRow() - 1, h.length).getValues()
      : [];

    const idxUnit  = h.indexOf('unitCode');
    const idxTahun = h.indexOf('tahun');
    const idxBulan = h.indexOf('bulan');
    const idxLast  = h.indexOf('lastNumber');
    const idxBy    = h.indexOf('updatedBy');
    const idxAt    = h.indexOf('updatedAt');

    const n  = new Date();
    const cy = n.getFullYear();
    const cm = String(n.getMonth() + 1).padStart(2, '0');

    // Cari baris untuk unit + tahun ini
    let targetRowNum = -1;
    let last = 0;

    for (let i = 0; i < rows.length; i++) {
      const rowUnit  = String(rows[i][idxUnit]  || '').toUpperCase().trim();
      const rowTahun = parseInt(rows[i][idxTahun] || 0);
      if (rowUnit === String(uc).toUpperCase() && rowTahun === cy) {
        targetRowNum = i + 2;
        last = parseInt(rows[i][idxLast] || 0);
        break;
      }
    }

    const nn  = last + 1;
    const usr = (uid && String(uid).trim()) ? String(uid).trim() : 'system';

    if (targetRowNum > 0) {
      // Update baris yang sudah ada
      const updateRow = s.getRange(targetRowNum, 1, 1, h.length).getValues()[0];
      updateRow[idxUnit]  = uc;
      updateRow[idxTahun] = cy;
      updateRow[idxBulan] = cm;
      updateRow[idxLast]  = nn;
      updateRow[idxBy]    = usr;
      updateRow[idxAt]    = new Date();
      s.getRange(targetRowNum, 1, 1, h.length).setValues([updateRow]);
    } else {
      // Buat baris baru untuk unit + tahun ini
      const newRow = h.map(col => {
        if (col === 'unitCode')   return uc;
        if (col === 'tahun')      return cy;
        if (col === 'bulan')      return cm;
        if (col === 'lastNumber') return nn;
        if (col === 'updatedBy')  return usr;
        if (col === 'updatedAt')  return new Date();
        return '';
      });
      s.appendRow(newRow);
    }

    SpreadsheetApp.flush();

    const kd = (di === "INTERNATIONAL") ? "22" : "21";
    return `${uc}.${kd}.${cy}.${cm}.${String(nn).padStart(4, '0')}`;

  } finally {
    lock.releaseLock();
  }
}

function validateFlight(rid, uid, manualKurs, usePpn, usePph, token) { 
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return { success: false, error: "Sistem sedang memproses validasi lain" };
  }
  try {
    // ── Baca SEMUA data yang dibutuhkan sekaligus di awal ──
    const ss      = SpreadsheetApp.getActiveSpreadsheet();
    const flSheet = ss.getSheetByName(CONFIG.SHEETS.FLIGHTS);
    const cfSheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_CONFIG);
    const rtSheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_RATE);
    const apSheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_AIRPORT_HOURS);
    const sgSheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_SIGNATORY);
    const exSheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_EXCHANGE_RATE);
    const sqSheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_SEQUENCE);

    // Baca semua sheet dalam satu batch
    const flH    = flSheet.getRange(1,1,1,flSheet.getLastColumn()).getValues()[0];
    const flRow  = flSheet.getRange(rid,1,1,flSheet.getLastColumn()).getValues()[0];
    const cfRows = cfSheet.getLastRow()>1 ? cfSheet.getRange(2,1,cfSheet.getLastRow()-1,cfSheet.getLastColumn()).getValues() : [];
    const rtRows = rtSheet.getLastRow()>1 ? rtSheet.getRange(2,1,rtSheet.getLastRow()-1,rtSheet.getLastColumn()).getValues() : [];
    const apRows = apSheet.getLastRow()>1 ? apSheet.getRange(2,1,apSheet.getLastRow()-1,apSheet.getLastColumn()).getValues() : [];
    const sgRows = sgSheet.getLastRow()>1 ? sgSheet.getRange(2,1,sgSheet.getLastRow()-1,sgSheet.getLastColumn()).getValues() : [];
    const exRows = exSheet.getLastRow()>1 ? exSheet.getRange(2,1,exSheet.getLastRow()-1,exSheet.getLastColumn()).getValues() : [];

    // ── Parse flight record ──
    const f = { rowId: rid };
    flH.forEach((k,j) => {
      let v = flRow[j];
      if (k==='flightDate') v = fmtDate(v);
      else if (k==='atd'||k==='ata') v = fmtTime(v);
      f[k] = v;
    });

    if (!f || f.isDeleted === true) return { success:false, error:'Data tidak ditemukan' };
    if (f.statusFlow !== CONFIG.STATUS.VALIDATION)
      return { success:false, error:'Status sudah bukan VALIDATION' };

    // ── Parse config ──
    const cfH = cfSheet.getRange(1,1,1,cfSheet.getLastColumn()).getValues()[0];
    const cfKi = cfH.indexOf('configKey'), cfVi = cfH.indexOf('configValue');
    const cfMap = {};
    cfRows.forEach(r => { if(r[cfKi]) cfMap[r[cfKi]] = r[cfVi]; });
    const vpPct = parseFloat(cfMap['vat_percent']) || 12;
    const ppPct = parseFloat(cfMap['pph_percent']) || 2;

    // ── Parse airport rule ──
    const apH   = apSheet.getRange(1,1,1,apSheet.getLastColumn()).getValues()[0];
    const apCi  = apH.indexOf('airportCode'), apNSi = apH.indexOf('normalStart');
    const apNEi = apH.indexOf('normalEnd'),   apMDi  = apH.indexOf('minDuration');
    const apAi  = apH.indexOf('isActive');
    const rule  = (() => {
      const r = apRows.find(r => r[apCi]===f.unitCode && r[apAi]!==false);
      if (!r) return null;
      return {
        normalStart: fmtTime(r[apNSi]),
        normalEnd:   fmtTime(r[apNEi]),
        minDuration: r[apMDi] ? parseInt(r[apMDi]) : 1
      };
    })();
    if (!rule) return { success:false, error:`Aturan bandara ${f.unitCode} tidak ditemukan` };

    // ── Parse rate ──
    const rtH  = rtSheet.getRange(1,1,1,rtSheet.getLastColumn()).getValues()[0];
    const rtUi = rtH.indexOf('unitCode'), rtSi = rtH.indexOf('service');
    const rtDi = rtH.indexOf('domInt'),   rtIi = rtH.indexOf('rateIDR');
    const rtUDi= rtH.indexOf('rateUSD'),  rtDli= rtH.indexOf('isDeleted');
    let category    = f.category;
    const existingDur = parseFloat(f.duration) || 0;
    let actualHours, billableHours;
    const minDur = rule.minDuration || 1;

    if (existingDur > 0) {
      actualHours   = existingDur;
      billableHours = Math.ceil(existingDur);
      if (billableHours < minDur) billableHours = minDur;
    } else {
      const waktu = f.dep_arr_loc==='DEP' ? f.atd : f.ata;
      const calc  = calcDuration(waktu, f.unitCode);
      if (calc.isNormal) return { success:false, error:'Penerbangan dalam jam normal' };
      category      = f.category || calc.category;
      actualHours   = calc.billableHours;
      billableHours = calc.billableHours;
    }

    const rObj = rtRows.find(r =>
      r[rtUi]===f.unitCode && r[rtSi]===category &&
      r[rtDi]===f.domInt   && r[rtDli]!==true
    );
    if (!rObj) return { success:false, error:`Rate tidak ditemukan: ${f.unitCode}/${category}/${f.domInt}` };

    // ── Hitung gross ──
    let gross, rUsed, kursUsedValue;
    if (f.domInt === 'DOMESTIK') {
      rUsed         = parseFloat(rObj[rtIi]) || 0;
      kursUsedValue = null;
      gross         = rUsed * billableHours;
    } else {
      let kurs = null;
      if (manualKurs && parseFloat(manualKurs) >= 1000) {
        kurs = parseFloat(manualKurs);
      } else {
        // Cari kurs dari exRows langsung tanpa memanggil fungsi terpisah
        const exHdr = exSheet.getRange(1,1,1,exSheet.getLastColumn()).getValues()[0];
        const exCi  = exHdr.indexOf('currency'), exRi = exHdr.indexOf('rate');
        const exUi  = exHdr.indexOf('updatedAt');
        const target = new Date(f.flightDate); target.setHours(0,0,0,0);
        for (const r of exRows) {
          if (String(r[exCi]).toUpperCase() !== 'USD') continue;
          const rd = new Date(r[exUi]); rd.setHours(0,0,0,0);
          if (rd.getTime() === target.getTime()) { kurs = parseFloat(r[exRi]); break; }
        }
      }
      if (!kurs || kurs < 1000) kurs = parseFloat(cfMap['default_kurs_usd']) || 15000;

      const rateUSD = parseFloat(rObj[rtH.indexOf('rateUSD')]) || 0;
      rUsed         = rateUSD;
      kursUsedValue = kurs;
      gross         = rateUSD * kurs * billableHours;

      // ── Simpan kurs tanpa LockService (kita sudah di dalam lock validateFlight) ──
      // TIDAK memanggil saveExchangeRateByDate() karena fungsi itu juga
      // memanggil LockService → deadlock. Simpan langsung ke sheet di sini.
      if (manualKurs && parseFloat(manualKurs) >= 1000) {
        try {
          _saveExchangeRateNoLock('USD', parseFloat(manualKurs), f.flightDate, uid);
        } catch(kursErr) {
          Logger.log('[validateFlight] Error simpan kurs (non-fatal): ' + kursErr.toString());
          // Non-fatal: validasi tetap lanjut
        }
      }
    }

    // ── Pajak ──
    const dppPpn = gross * (11/12);
    const vat    = (usePpn !== false) ? dppPpn * vpPct / 100 : 0;
    const pph    = (usePph !== false) ? gross * ppPct / 100 : 0;
    const total  = gross + vat - pph;

    // ── Generate nomor invoice ──
    const invNo = genInvoiceNo(f.unitCode, f.domInt, uid);

    // ── Ambil signatory langsung dari rows ──
    const sgH   = sgSheet.getRange(1,1,1,sgSheet.getLastColumn()).getValues()[0];
    const sgUi  = sgH.indexOf('unitCode'), sgNi = sgH.indexOf('name');
    const sgPi  = sgH.indexOf('position'), sgNpi= sgH.indexOf('nip');
    const sgAi  = sgH.indexOf('isActive');
    const sgRow = sgRows.find(r =>
      (r[sgUi]===f.unitCode || r[sgUi]==='ALL') && r[sgAi]!==false
    );
    // ✅ FIX: sertakan qrcodeBase64 di snapshot — dokumen yang sudah
    // divalidasi TIDAK BOLEH lagi bergantung pada Master_Signatory yang
    // bisa berubah (ganti orang, ganti QR, dsb). Snapshot harus final.
    const sgB64i = sgH.indexOf('qrcodeBase64');
    const signatorySnap = sgRow ? JSON.stringify({
      name:          sgRow[sgNi]  || '',
      position:      sgRow[sgPi]  || '',
      nip:           sgRow[sgNpi] || '',
      qrcodeBase64:  (sgB64i !== -1 ? sgRow[sgB64i] : '') || '',
      unitCode:      sgRow[sgUi]  || '',
    }) : '';

    // ── Simpan ke sheet dalam satu operasi ──
    const snapshot = {
      category, duration: actualHours,
      statusFlow:        CONFIG.STATUS.INVOICED,
      invoiceNo:         invNo,
      grossAmount:       gross,
      vatAmount:         vat,
      pphAmount:         pph,
      totalAmount:       total,
      rateUsed:          parseFloat(rUsed),
      kursUsed:          kursUsedValue ? parseFloat(kursUsedValue) : '',
      validatedAt:       new Date().toISOString(),
      validatedBy:       uid,
      vatPctSnapshot:    vpPct,
      pphPctSnapshot:    ppPct,
      signatorySnapshot: signatorySnap,
    };

    updateFlight(rid, snapshot, uid, 'VALIDATE');

    return {
      success:     true,
      invoiceNo:   invNo,
      totalAmount: total,
      grossAmount: gross,
      vatAmount:   vat,
      pphAmount:   pph,
      kursUsed:    kursUsedValue,
      category,
      duration:    actualHours,
      // ✅ FIX: sertakan snapshot signatory & waktu validasi agar client bisa
      // langsung memperbarui state lokal — tanpa ini, Preview/View yang
      // dibuka SEBELUM refresh browser tidak menemukan QR & nama penandatangan
      // karena field ini tidak pernah dikirim balik ke client.
      signatorySnapshot: snapshot.signatorySnapshot,
      validatedAt:       snapshot.validatedAt,
      validatedBy:       uid,
    };

  } catch(err) {
    Logger.log('validateFlight error: ' + err.toString());
    return { success:false, error:'Kesalahan sistem: ' + err.toString() };
  } finally {
    lock.releaseLock();
  }
}

// ==================== RATE LIMITING ====================
const RATE_LIMIT = {
  LOGIN_MAX_ATTEMPTS: 5,           // Max 5 percobaan gagal
  LOGIN_WINDOW_MINUTES: 15,        // Dalam 15 menit
  LOCKOUT_MINUTES: 30,             // Lockout 30 menit
  API_MAX_CALLS_PER_MINUTE: 60,    // Max 60 call/menit per user
};

/**
 * Catat percobaan login gagal
 */
function logFailedLoginAttempt(username) {
  try {
    const cache = CacheService.getScriptCache();
    const key = 'failed_login_' + String(username).toLowerCase().replace(/[^a-z0-9]/g, '_');
    
    const current = parseInt(cache.get(key) || '0', 10);
    cache.put(key, String(current + 1), RATE_LIMIT.LOGIN_WINDOW_MINUTES * 60);
    
    // Jika mencapai threshold, lock account sementara
    if (current + 1 >= RATE_LIMIT.LOGIN_MAX_ATTEMPTS) {
      const lockKey = 'locked_' + String(username).toLowerCase().replace(/[^a-z0-9]/g, '_');
      cache.put(lockKey, '1', RATE_LIMIT.LOCKOUT_MINUTES * 60);
      logAudit('system', 'ACCOUNT_LOCKED', 'Auth', null, null, null, null, null, username);
    }
  } catch (e) {
    Logger.log('[RATE_LIMIT] logFailedLoginAttempt error: ' + e.toString());
  }
}

/**
 * Simpan kurs tanpa LockService — HANYA dipanggil dari dalam fungsi
 * yang sudah memegang lock (mis. validateFlight).
 * Menghindari deadlock nested lock di Apps Script.
 */
function _saveExchangeRateNoLock(currency, rate, flightDate, uid) {
  const s = SpreadsheetApp.getActiveSpreadsheet()
              .getSheetByName(CONFIG.SHEETS.MASTER_EXCHANGE_RATE);
  if (!s) return;

  const h   = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
  const ci  = h.indexOf('currency');
  const ri  = h.indexOf('rate');
  const si  = h.indexOf('source');
  const ui  = h.indexOf('updatedAt');
  const ubi = h.indexOf('updatedBy');
  if (ci === -1 || ri === -1 || ui === -1) return;

  const target = new Date(flightDate);
  target.setHours(0, 0, 0, 0);

  // Cari baris yang sudah ada untuk currency + tanggal yang sama
  let existingRowNum = -1;
  if (s.getLastRow() >= 2) {
    const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
    for (let i = 0; i < rows.length; i++) {
      if (String(rows[i][ci]).toUpperCase().trim() !== currency.toUpperCase()) continue;
      const rawDate = rows[i][ui];
      if (!rawDate) continue;
      const rowDate = rawDate instanceof Date ? new Date(rawDate) : new Date(rawDate);
      rowDate.setHours(0, 0, 0, 0);
      if (rowDate.getTime() === target.getTime()) {
        existingRowNum = i + 2;
        break;
      }
    }
  }

  if (existingRowNum > 0) {
    const rowData = s.getRange(existingRowNum, 1, 1, h.length).getValues()[0];
    rowData[ri] = rate;
    rowData[ui] = target;
    if (si  !== -1) rowData[si]  = 'Validasi';
    if (ubi !== -1) rowData[ubi] = uid || 'system';
    s.getRange(existingRowNum, 1, 1, h.length).setValues([rowData]);
  } else {
    const newRow = h.map(col => {
      if (col === 'currency')  return currency.toUpperCase();
      if (col === 'rate')      return rate;
      if (col === 'source')    return 'Validasi';
      if (col === 'updatedAt') return target;
      if (col === 'updatedBy') return uid || 'system';
      return '';
    });
    s.appendRow(newRow);
  }
  logAudit(uid, 'UPSERT', CONFIG.SHEETS.MASTER_EXCHANGE_RATE, null,
           'rate', null, rate, null, 'no-lock write from validateFlight');
}

function validateFlightRecord(rid, manualKurs, usePpn, usePph, username, token) {
  try {
    if (!username) username = Session.getActiveUser().getEmail();
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };
    
    // VALIDASI HANYA boleh dilakukan oleh ADMIN
    if (!ctx.isAdmin) {
      logAudit(username, 'UNAUTHORIZED_VALIDATE_ATTEMPT', 'Flights', rid);
      return { success: false, error: 'Hanya administrator yang dapat memvalidasi flight' };
    }
    
    const flight = getFlightByRowId(rid);
    if (!flight) return { success: false, error: 'Data tidak ditemukan' };
    
    // Extra validation: kurs manual harus dalam rentang wajar
    if (manualKurs && parseFloat(manualKurs) > 0) {
      const kursNum = parseFloat(manualKurs);
      if (kursNum < 1000 || kursNum > 100000) {
        return { success: false, error: 'Kurs tidak wajar (harus 1,000 - 100,000 IDR/USD)' };
      }
    }
    
    const ppn = (usePpn === undefined || usePpn === null) ? true : Boolean(usePpn);
    const pph = (usePph === undefined || usePph === null) ? true : Boolean(usePph);
    return validateFlight(rid, ctx.username, manualKurs, ppn, pph, token);
  } catch(e) {
    return { success: false, error: e.toString() };
  }
}

function getFlightByRowId(rid) {
  rid = parseInt(rid, 10);
  const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.FLIGHTS);
  if (!s || !rid || isNaN(rid) || rid < 2 || rid > s.getLastRow()) return null;
  const h   = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
  const row = s.getRange(rid, 1, 1, s.getLastColumn()).getValues()[0];
  const o = { rowId: rid };
  h.forEach((k, j) => {
    let v = row[j];
    if (k === "flightDate") v = fmtDate(v);
    else if (k === "atd" || k === "ata") v = fmtTime(v);
    else v = sv(v); // ✅ FIX: sanitasi Date/undefined/null → string aman untuk RPC
    o[k] = v;
  });
  return o;
}

/**
 * ✅ OPTIMASI: baca beberapa flight sekaligus dengan SATU getRange
 * (rentang rowId terkecil→terbesar), lalu pilih yang diminta dari Map.
 * Mengganti pola `rowIds.map(rid => getFlightByRowId(rid))` yang membaca
 * sheet N kali (N = jumlah rowId).
 */
function getFlightsByRowIds(rowIds) {
  const ids = [...new Set(rowIds.map(Number).filter(n => !isNaN(n) && n >= 2))];
  if (!ids.length) return [];
  const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.FLIGHTS);
  if (!s || s.getLastRow() < 2) return [];
  const h   = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
  const minR = Math.min(...ids);
  const maxR = Math.min(Math.max(...ids), s.getLastRow());
  if (maxR < minR) return [];
  const rows = s.getRange(minR, 1, maxR - minR + 1, s.getLastColumn()).getValues();
  const result = [];
  for (let k = 0; k < rows.length; k++) {
    const rowId = minR + k;
    if (!ids.includes(rowId)) continue;
    const o = { rowId: rowId };
    h.forEach((col, j) => {
      let v = rows[k][j];
      if (col === "flightDate") v = fmtDate(v);
      else if (col === "atd" || col === "ata") v = fmtTime(v);
      else v = sv(v);
      o[col] = v;
    });
    result.push(o);
  }
  return result;
}

    return { success: false, error: "Hanya INVOICED yang bisa ke UNPAID" };
  if (ns === CONFIG.STATUS.PAID && os !== CONFIG.STATUS.UNPAID)
    return { success: false, error: "Hanya UNPAID yang bisa ke PAID" };

  const extraFields = {};
  if (ns === CONFIG.STATUS.UNPAID) {
    extraFields.receiptDate = new Date().toISOString();
    Logger.log('[DEBUG] receiptDate akan disimpan: ' + extraFields.receiptDate);
    Logger.log('[DEBUG] extraFields: ' + JSON.stringify(extraFields));
  }

  Logger.log('[DEBUG] updateFlight dipanggil dengan: ' + JSON.stringify({ statusFlow: ns, ...extraFields }));
  const result = updateFlight(rid, { statusFlow: ns, ...extraFields }, uid, false);
  Logger.log('[DEBUG] hasil updateFlight: ' + JSON.stringify(result));
  
  logAudit(uid, "STATUS_CHANGE", "Flights", rid, "statusFlow", os, ns);
  return { success: true, oldStatus: os, newStatus: ns };
}

  }
}

// ==================== PASSWORD HASHING (PBKDF2-SHA256) ====================
// Apps Script tidak punya bcrypt native, gunakan PBKDF2 via Utilities
const PBKDF2_ITERATIONS = 12000;
const PBKDF2_SALT_LENGTH = 16;

/**
 * Hash password dengan PBKDF2-SHA256 + random salt
 * Format output: "pbkdf2$iterations$salt$hash" (portable, self-describing)
 */
function hashPassword(plainPassword) {
  if (!plainPassword || typeof plainPassword !== 'string') {
    throw new Error('Password tidak valid');
  }
  if (plainPassword.length < 8) {
    throw new Error('Password minimal 8 karakter');
  }
  if (plainPassword.length > 128) {
    throw new Error('Password maksimal 128 karakter');
  }
  
  // ✅ FIX: Generate random salt menggunakan digest SHA-256 dari UUID + timestamp
  const rawBytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256, 
    Utilities.getUuid() + Date.now() + Math.random()
  );
  const saltBytes = rawBytes.slice(0, PBKDF2_SALT_LENGTH);
  const saltB64 = Utilities.base64Encode(saltBytes);
  
  // Derive key menggunakan HMAC berulang (PBKDF2 manual)
  const derivedKey = pbkdf2Sha256(plainPassword, saltBytes, PBKDF2_ITERATIONS, 32);
  const hashB64 = Utilities.base64Encode(derivedKey);
  
  return `pbkdf2$${PBKDF2_ITERATIONS}$${saltB64}$${hashB64}`;
}

/**
 * Verifikasi password terhadap hash yang tersimpan
 * Constant-time comparison untuk mencegah timing attack
 */
function verifyPassword(plainPassword, storedHash) {
  if (!plainPassword || !storedHash) return false;
  
  const parts = storedHash.split('$');
  if (parts.length !== 4 || parts[0] !== 'pbkdf2') {
    // Legacy plain text password — force migration
    return plainPassword === storedHash;
  }
  
  const iterations = parseInt(parts[1], 10);
  const saltB64 = parts[2];
  const expectedHashB64 = parts[3];
  
  try {
    const saltBytes = Utilities.base64Decode(saltB64);
    const derivedKey = pbkdf2Sha256(plainPassword, saltBytes, iterations, 32);
    const computedHashB64 = Utilities.base64Encode(derivedKey);
    
    // Constant-time string comparison
    return constantTimeEquals(computedHashB64, expectedHashB64);
  } catch (e) {
    Logger.log('[AUTH] verifyPassword error: ' + e.toString());
    return false;
  }
}

/**
 * PBKDF2 implementation manual (Apps Script tidak punya native)
 */
function pbkdf2Sha256(password, salt, iterations, keyLen) {
  const passwordBytes = Utilities.newBlob(password).getBytes();
  let result = new Array(keyLen).fill(0);
  
  const blocksNeeded = Math.ceil(keyLen / 32);
  for (let block = 1; block <= blocksNeeded; block++) {
    // U1 = HMAC(password, salt || INT_32_BE(block))
    const blockBytes = [
      (block >> 24) & 0xff, (block >> 16) & 0xff,
      (block >> 8) & 0xff, block & 0xff
    ];

    const saltWithBlock = salt.concat(blockBytes);
    
    // ✅ FIX: Konversi byte array ke binary string (bukan UTF-8)
    const saltStr = saltWithBlock.map(b => String.fromCharCode(b)).join('');
    
    let u = Utilities.computeHmacSha256Signature(saltStr, password);
    u = u.map(b => b < 0 ? b + 256 : b);
    
    let xored = u.slice();
    
    for (let i = 1; i < iterations; i++) {
      // ✅ FIX: Konversi hasil HMAC sebelumnya ke binary string
      const uStr = u.map(b => String.fromCharCode(b)).join('');
      u = Utilities.computeHmacSha256Signature(uStr, password).map(b => b < 0 ? b + 256 : b);
      
      for (let j = 0; j < 32; j++) {
        xored[j] ^= u[j];
      }
    }
    
    for (let j = 0; j < 32 && (block - 1) * 32 + j < keyLen; j++) {
      result[(block - 1) * 32 + j] = xored[j];
    }
  }
  
  return result;
}

/**
 * Constant-time string comparison — mencegah timing attack
 */
function constantTimeEquals(a, b) {
  if (typeof a !== 'string' || typeof b !== 'string') return false;
  if (a.length !== b.length) return false;
  
  let result = 0;
  for (let i = 0; i < a.length; i++) {
    result |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }
  return result === 0;
}

function getSessionSecret() {
  const sp = PropertiesService.getScriptProperties();
  let secret = sp.getProperty('SESSION_SECRET');
  
  if (!secret) {

    secret = (Utilities.getUuid() + Utilities.getUuid() + Utilities.getUuid())

               .substring(0, 64);

  }
  
  return secret;
}

// ==================== AUTH ====================
function doLoginCheck(username, password) {
  const lock = LockService.getScriptLock();
  lock.waitLock(5000);
  try {
    if (!username || !password) {
      return { success: false, error: 'Username dan password wajib diisi' };
    }
    
    // Sanitasi username
    username = String(username).trim().toLowerCase();
    if (username.length < 3 || username.length > 64) {
      return { success: false, error: 'Format username tidak valid' };
    }
    if (!/^[a-z0-9._@-]+$/.test(username)) {
      return { success: false, error: 'Username mengandung karakter tidak valid' };
    }
    
    // ── CHECK RATE LIMIT ──
    if (isAccountLocked(username)) {
      logAudit('system', 'LOCKED_LOGIN_ATTEMPT', 'Auth', null);
      return { 
        success: false, 
        error: `Akun dikunci sementara karena terlalu banyak percobaan gagal. Coba lagi dalam ${RATE_LIMIT.LOCKOUT_MINUTES} menit.`,
        locked: true
      };
    }
    
    const failedAttempts = getFailedAttempts(username);
    
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.MASTER_USER);
    if (!s) return { success: false, error: 'Sheet Master_User tidak ditemukan' };

    const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getRange(2, 1, Math.max(1, s.getLastRow() - 1),
                            s.getLastColumn()).getValues();

    const ui = h.indexOf('username');
    const pi = h.indexOf('passwordHash');
    const ai = h.indexOf('isActive');
    const ni = h.indexOf('fullName');
    const ci = h.indexOf('unitCode');

    const rowIdx = rows.findIndex(r =>
      String(r[ui]).toLowerCase() === username
    );
    
    if (rowIdx === -1) {
      // Delay artifisial untuk mencegah username enumeration
      Utilities.sleep(500 + Math.random() * 500);
      logFailedLoginAttempt(username);
      return { 
        success: false, 
        error: 'Username atau password salah',
        attemptsRemaining: Math.max(0, RATE_LIMIT.LOGIN_MAX_ATTEMPTS - failedAttempts - 1)
      };
    }
    
    const row = rows[rowIdx];
    
    if (ai !== -1 && !row[ai]) {
      return { success: false, error: 'Akun tidak aktif — hubungi administrator' };
    }
    
    // Verifikasi password
    if (!verifyPassword(password, String(row[pi] || ''))) {
      logFailedLoginAttempt(username);
      const remaining = Math.max(0, RATE_LIMIT.LOGIN_MAX_ATTEMPTS - failedAttempts - 1);
      
      if (remaining === 0) {
        return { 
          success: false, 
          error: `Terlalu banyak percobaan gagal. Akun dikunci selama ${RATE_LIMIT.LOCKOUT_MINUTES} menit.`,
          locked: true
        };
      }
      
      return { 
        success: false, 
        error: `Username atau password salah. ${remaining} percobaan tersisa.`,
        attemptsRemaining: remaining
      };
    }
    
    // ── LOGIN SUKSES — reset counter ──
    resetLoginAttempts(username);

    // Auto-migrate legacy plain text password → hash baru
    // + auto-DOWNGRADE iterasi PBKDF2 lama (100k) ke iterasi baru yang lebih ringan,
    //   agar login SELANJUTNYA untuk akun lama langsung jauh lebih cepat.
    const storedHash = String(row[pi] || '');
    const storedIterations = storedHash.startsWith('pbkdf2$')
      ? (parseInt(storedHash.split('$')[1], 10) || 0)
      : 0;
    const needsRehash = !storedHash.startsWith('pbkdf2$')
      || storedIterations > PBKDF2_ITERATIONS * 1.5;

    if (needsRehash) {
      try {
        const newHash = hashPassword(password);
        row[pi] = newHash;
        rows[rowIdx][pi] = newHash;
        s.getRange(rowIdx + 2, 1, 1, h.length).setValues([row]);
        Logger.log('[AUTH] Password rehashed (iterasi diperbarui) untuk: ' + username);
      } catch (e) {
        Logger.log('[AUTH] Rehash gagal: ' + e.toString());
      }
    }

    const sp         = PropertiesService.getScriptProperties();
    const sessionKey = 'SESSION_' + username.toUpperCase();

    // ── Bersihkan session lama untuk username ini (apapun kondisinya) ──
    // Jika ada session lama yang belum dihapus (crash/force close),
    // kita validasi dengan ketat: hanya blokir jika BENAR-BENAR masih aktif
    const existing = sp.getProperty(sessionKey);
    if (existing) {
      try {
        const sess    = JSON.parse(existing);
        const now     = Date.now();
        const expired = now > (sess.expiresAt || 0);
        const tooOld  = (now - (sess.loginAt || 0)) > 8 * 60 * 60 * 1000;

        if (expired || tooOld) {
          // Expired → hapus otomatis, izinkan login
          sp.deleteProperty(sessionKey);
          Logger.log('[AUTH] Auto-cleared expired/old session for: ' + username);
        } else {
          // Benar-benar masih aktif
          const ageMin = Math.floor((now - sess.loginAt) / 60000);
          const loginTime = new Date(sess.loginAt)
            .toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' });
          return {
            success:    false,
            error:      `Akun masih aktif sejak ${loginTime} (${ageMin} menit lalu).`,
            concurrent: true,
            canForce:   true,
            sessionAge: ageMin
          };
        }
      } catch (e) {
        // JSON corrupt → hapus & lanjut
        sp.deleteProperty(sessionKey);
        Logger.log('[AUTH] Cleared corrupt session for: ' + username);
      }
    }

    // ── Buat session baru ──
    const now   = Date.now();
    const secret = getSessionSecret();  // ✅ GANTI INI
    const token = Utilities.base64Encode(
      Utilities.computeHmacSha256Signature(
        username + now + Math.random(),
        secret   // ✅ Gunakan dari ScriptProperties
      )
    ).replace(/[^a-zA-Z0-9]/g, '').substring(0, 32);

    const userUnit = (ci !== -1 && row[ci])
      ? String(row[ci]).toUpperCase().trim() : 'ALL';

    const sessionData = JSON.stringify({
      token,
      username,
      fullName:  String(row[ni] || username),
      unitCode:  userUnit,
      loginAt:   now,
      expiresAt: now + 8 * 60 * 60 * 1000
    });

    sp.setProperty(sessionKey, sessionData);
    Logger.log('[AUTH] New session created for: ' + username + ' token: ' + token.substring(0,8));

    logAudit(username, 'LOGIN', 'Auth', null);

    return {
      success:  true,
      fullName: String(row[ni] || username),
      unitCode: userUnit,
      token,
      sessionKey  // kembalikan ke client untuk dipakai saat logout
    };

  } catch (e) {
    Logger.log('[AUTH] doLoginCheck error: ' + e.toString());
    return { success: false, error: 'Terjadi kesalahan server: ' + e.message };
  } finally {
    lock.releaseLock();
  }
}

/**
 * checkSession — validasi dari ScriptProperties (bukan UserProperties)
 * Menerima token dari client untuk verifikasi yang akurat.
 */
function checkSession(username, token) {
  try {
    // Jika tidak ada username → session tidak valid
    if (!username) return { loggedIn: false };

    const sp         = PropertiesService.getScriptProperties();
    const sessionKey = 'SESSION_' + String(username).toUpperCase();
    const raw        = sp.getProperty(sessionKey);
    if (!raw) return { loggedIn: false };

    const sess = JSON.parse(raw);
    const now  = Date.now();

    // Cek expired
    if (now > (sess.expiresAt || 0)) {
      sp.deleteProperty(sessionKey); // auto-cleanup
      return { loggedIn: false };
    }

    // Token WAJIB cocok — sebelumnya token kosong melewati pemeriksaan ini sama sekali
    if (sess.token !== token) {
      return { loggedIn: false };
    }

    return {
      loggedIn: true,
      username: sess.username  || username,
      fullName: sess.fullName  || username,
      unitCode: sess.unitCode  || 'ALL'
    };

  } catch(e) {
    Logger.log('[AUTH] checkSession error: ' + e.toString());
    return { loggedIn: false };
  }
}

/**
 * doLogout — menerima username eksplisit dari client
 * agar tidak bergantung pada UserProperties yang tidak reliable
 */
function doLogout(username, token) {
  try {
    const sp = PropertiesService.getScriptProperties();

    if (username) {
      const sessionKey = 'SESSION_' + String(username).toUpperCase();
      // Verifikasi token sebelum menghapus sesi — mencegah user lain memaksa
      // logout akun orang lain hanya dengan mengetahui username-nya.
      const raw = sp.getProperty(sessionKey);
      if (raw) {
        try {
          const sess = JSON.parse(raw);
          if (sess.token !== token) {
            logAudit(username, 'UNAUTHORIZED_LOGOUT_ATTEMPT', 'Auth', null);
            return { success: false, error: 'Token tidak valid' };
          }
        } catch(e) {}
      }
      sp.deleteProperty(sessionKey);
      Logger.log('[AUTH] Session deleted for: ' + username);
      logAudit(username, 'LOGOUT', 'Auth', null, 'session', sessionKey, 'deleted');
    } else {
      // Fallback: coba baca dari UserProperties
      try {
        const up  = PropertiesService.getUserProperties();
        const usr = up.getProperty('SESSION_USER');
        if (usr) {
          sp.deleteProperty('SESSION_' + usr.toUpperCase());
          Logger.log('[AUTH] Session deleted (fallback) for: ' + usr);
          logAudit(username, 'LOGOUT', 'Auth', null);
        }
        up.deleteAllProperties();
      } catch (e2) {
        Logger.log('[AUTH] Fallback logout error: ' + e2);
      }
    }

    // Bersihkan UserProperties
    try {
      PropertiesService.getUserProperties().deleteAllProperties();
    } catch(e3) {}

    return { success: true };
  } catch (e) {
    Logger.log('[AUTH] doLogout error: ' + e.toString());
    return { success: true }; // tetap sukses agar client bisa clear state
  }
}

/**
 * Bersihkan semua session yang sudah expired.
 * Bisa dipanggil manual dari Master Settings atau otomatis.
 */
function cleanExpiredSessions() {
  try {
    const sp    = PropertiesService.getScriptProperties();
    const all   = sp.getProperties();
    const now   = Date.now();
    let cleaned = 0;

    Object.keys(all).forEach(key => {
      if (!key.startsWith('SESSION_')) return;
      try {
        const sess = JSON.parse(all[key]);
        // Hapus jika expired ATAU lebih dari 9 jam (safety)
        if (!sess.expiresAt || now > sess.expiresAt || (now - sess.loginAt) > 9 * 3600 * 1000) {
          sp.deleteProperty(key);
          cleaned++;
          Logger.log(`[AUTH] Cleaned expired session: ${key}`);
        }
      } catch(e) {
        // Session corrupt → hapus
        sp.deleteProperty(key);
        cleaned++;
      }
    });

    return { success: true, cleaned };
  } catch(e) {
    return { success: false, error: e.toString() };
  }
}

    return { success: false, error: e.toString() };
  }

/**
 * Generate token akses publik untuk 1 flight record (invoice+kwitansi 2-halaman).
 * Dipanggil saat admin klik "Kirim WhatsApp". Token tidak bisa dipakai
 * mengakses data lain, read-only, expired otomatis.

function generateDocShareToken(rowId, uid, token) {

    '/doc.html?rid=' + encodeURIComponent(rowId) + '&t=' + encodeURIComponent(shareToken);

// ==================== EMAIL SENDING ====================
/**
 * Kirim email berisi ringkasan tagihan/dokumen + link download ke
 * alamat email operator. Dipanggil dari modal "Kirim via Email"
 * (kanal alternatif dari tombol Send, selain WhatsApp).
 *
 * @param {string} toEmail  - alamat tujuan
 * @param {string} subject  - subjek email
 * @param {string} bodyText - isi email (plain text)
 * @param {string} username - username pengirim (untuk audit log)
 * @param {string} token    - session token
 */
function sendInvoiceEmail(toEmail, subject, bodyText, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };

    const email = String(toEmail || '').trim();
    if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      return { success: false, error: 'Alamat email tidak valid' };
    }
    if (!subject || !bodyText) {
      return { success: false, error: 'Subjek dan isi email wajib diisi' };
    }

    // ✅ FIX: getRemainingDailyQuota() butuh scope yang sama dengan
    // sendEmail() — jika scope belum diotorisasi, pemanggilan ini sendiri
    // bisa melempar exception SEBELUM sempat mengecek sisa kuota. Bungkus
    // terpisah agar kegagalan di sini tidak menghentikan percobaan kirim
    // yang sebenarnya (dan agar pesan error permission tetap jelas, bukan
    // tertutupi oleh pre-check ini).
    try {
      const remaining = MailApp.getRemainingDailyQuota();
      Logger.log('[sendInvoiceEmail] Sisa kuota harian: ' + remaining);
      if (remaining <= 0) {
        return {
          success: false,
          error: 'Kuota pengiriman email harian akun pengirim sudah habis ' +
                 '(reset otomatis besok pagi). Gunakan kanal WhatsApp untuk sementara.'
        };
      }
    } catch (quotaErr) {
      Logger.log('[sendInvoiceEmail] Gagal cek kuota (non-fatal, lanjut coba kirim): ' + quotaErr);
      // Lanjutkan ke sendEmail() — jika memang scope belum ada, akan
      // tertangkap dengan pesan yang jelas di catch block utama di bawah.
    }

    MailApp.sendEmail({
      to:      email,
      subject: String(subject).trim().substring(0, 200),
      body:    String(bodyText)
    });

    logAudit(ctx.username, 'SEND_EMAIL', 'Flights', null, null, null, null, null,
      'To: ' + email + ' | Subject: ' + subject);

    return { success: true };
  } catch (e) {
    Logger.log('[sendInvoiceEmail] error: ' + e.toString());
    // ✅ FIX: urutan pengecekan diperbaiki — "permission"/"scope" dicek
    // LEBIH DULU karena pesan permission Apps Script sering memuat kata
    // "Quota" di dalam NAMA METHOD (mis. getRemainingDailyQuota), yang
    // sebelumnya salah tertangkap oleh pola /quota/i dan membuat error izin
    // terdiagnosis keliru sebagai "kuota habis".
    const raw = e.toString();
    let friendly = raw;
    if (/not sufficient|insufficient permission|required permission|authorization is required/i.test(raw)) {
      friendly = 'Izin pengiriman email belum diaktifkan untuk aplikasi ini. ' +
                 'Admin perlu membuka appsscript.json, menambahkan scope ' +
                 '"https://www.googleapis.com/auth/script.send_mail", lalu membuat ' +
                 'deployment baru dan menyetujui izin saat diminta.';
    } else if (/too many times for one day|daily limit/i.test(raw)) {
      friendly = 'Kuota pengiriman email harian sudah habis. Coba lagi besok atau gunakan WhatsApp.';
    } else if (/Invalid email/i.test(raw)) {
      friendly = 'Alamat email tujuan tidak valid atau ditolak oleh server Google.';
    }
    return { success: false, error: friendly };
  }
}

const BATCH_TOKEN_VALID_DAYS = 60;

/**
 * Helper internal: ambil satu baris Doc_Batch berdasarkan batchId.
 * Dipakai oleh guard di softDeleteFlight, voidInvoice, dan generateBatchInvoiceToken.
 */
function _getBatchById(batchId) {
  if (!batchId) return null;
  try {
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    if (!s || s.getLastRow() < 2) return null;
    const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
    const bi   = h.indexOf('batchId');
    const si   = h.indexOf('status');
    if (bi === -1) return null;
    for (let i = 0; i < rows.length; i++) {
      if (String(rows[i][bi]).trim() === String(batchId).trim()) {
        const obj = { rowId: i + 2 };
        h.forEach((k, j) => { obj[k] = rows[i][j]; });
        return obj;
      }
    }
    return null;
  } catch(e) {
    Logger.log('[_getBatchById] error: ' + e.toString());
    return null;
  }
}

/**
 * Ambil status link batch untuk sekumpulan flight sekaligus.
 * Dipakai klien untuk:
 *   - Menandai baris data yang sudah punya link batch (indikator UI).
 *   - Menampilkan di modal kirim batch mana yang sudah tergenerate.
 *   - Mengirim ulang (re-send) link yang sudah ada tanpa regenerate.
 *
 * Mengembalikan map rowId -> { batchId, status, portalUrl, expiresAt, operatorName, grandTotal }.
 * Hanya batch yang masih valid (tidak revoked, status SENT/PROOF_SUBMITTED/REJECTED) dihitung;
 * batch VERIFIED sudah lunas jadi dianggap "selesai" (tetap disertakan dengan flag done=true).
 *
 * @param {number[]} rowIds   - Array rowId flight
 * @param {string}   username - Username admin
 * @param {string}   token    - Session token
 */
function getFlightBatchLinks(rowIds, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired', data: {} };
    if (!ctx.isAdmin) {
      return { success: false, error: 'Hanya admin yang dapat memeriksa batch', data: {} };
    }
    if (!Array.isArray(rowIds) || rowIds.length === 0) {
      return { success: true, data: {} };
    }
    const ids = [...new Set(rowIds.map(Number).filter(n => !isNaN(n) && n >= 2))];
    if (ids.length === 0) return { success: true, data: {} };

    // Ambil batchId per flight
    const flights = getFlightsByRowIds(ids);
    const batchMap = {}; // batchId -> { ...batch }
    flights.forEach(f => {
      if (f.batchId && String(f.batchId).trim()) {
        if (!batchMap[String(f.batchId).trim()]) {
          batchMap[String(f.batchId).trim()] = null;
        }
      }
    });
    const batchIds = Object.keys(batchMap);
    if (batchIds.length === 0) return { success: true, data: {} };

    // Baca sheet Doc_Batch sekali
    const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    let batchRows = {};
    if (s && s.getLastRow() >= 2) {
      const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
      const bi   = h.indexOf('batchId');
      const data = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
      for (let i = 0; i < data.length; i++) {
        const bid = String(data[i][bi] || '').trim();
        if (!bid || !batchIds.includes(bid)) continue;
        const obj = { rowId: i + 2 };
        h.forEach((k, j) => { obj[k] = data[i][j]; });
        batchRows[bid] = obj;
      }
    }

    const baseUrl = CONFIG.PUBLIC_BASE_URL || ScriptApp.getService().getUrl();
    const result = {};
    let rowIdToBatch = {};
    flights.forEach(f => {
      if (!f.batchId || !String(f.batchId).trim()) return;
      const bid = String(f.batchId).trim();
      if (!rowIdToBatch[bid]) rowIdToBatch[bid] = [];
      rowIdToBatch[bid].push(f.rowId);
    });

    Object.keys(batchRows).forEach(bid => {
      const b = batchRows[bid];
      if (b.revoked === true) return; // link dicabut → dianggap tidak ada
      const memberIds = rowIdToBatch[bid] || [];
      result[bid] = {
        batchId:     bid,
        status:      String(b.status || ''),
        operatorName:b.operatorName || '',
        airlineCode: b.airlineCode || '',
        grandTotal:  parseFloat(b.grandTotal) || 0,
        expiresAt:   (b.expiresAt instanceof Date ? b.expiresAt : new Date(b.expiresAt || 0)).toISOString(),
        portalUrl:   baseUrl + '/batch.html?bid=' + encodeURIComponent(bid) + '&t=' + encodeURIComponent(b.token || ''),
        memberRowIds: memberIds,
        done:        String(b.status || '').toUpperCase() === 'VERIFIED'
      };
    });

    return _deepSanitizeForClient({ success: true, data: result });
  } catch(e) {
    Logger.log('[getFlightBatchLinks] error: ' + e.toString());
    return { success: false, error: e.toString(), data: {} };
  }
}

/**
 * Generate token batch invoice untuk sejumlah rowId flight sekaligus.
 * Dipanggil dari frontend saat admin centang beberapa data di tab Active Invoice
 * lalu klik "Kirim Tagihan Batch".
 *
 * Validasi:
 *   1. Semua rowId harus berstatus INVOICED.
 *   2. Tidak boleh ada rowId yang sudah terikat batch SENT/PROOF_SUBMITTED lain.
 *   3. Semua rowId harus dari unit yang sama (satu operator per batch).
 *
 * @param {number[]} rowIds   - Array rowId flight yang dipilih
 * @param {string}   username - Username admin
 * @param {string}   token    - Session token
 */
function generateBatchInvoiceToken(rowIds, username, token) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return { success: false, error: 'Sistem sedang memproses permintaan lain, coba lagi.' };
  }
  try {
    // ── Otorisasi ──
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };
    if (!ctx.isAdmin) {
      logAudit(username, 'UNAUTHORIZED_BATCH_CREATE', 'Doc_Batch', null);
      return { success: false, error: 'Hanya admin yang dapat membuat batch invoice' };
    }

    // ── Validasi input ──
    if (!Array.isArray(rowIds) || rowIds.length < 1) {
      return { success: false, error: 'Pilih minimal 1 data untuk batch' };
    }
    if (rowIds.length > 20) {
      return { success: false, error: 'Maksimal 20 data per batch' };
    }

    // ── Baca semua flight yang diminta sekaligus (1 getRange) ──
    const flights = getFlightsByRowIds(rowIds);
    if (flights.length !== rowIds.length) {
      return { success: false, error: 'Beberapa data tidak ditemukan atau sudah dihapus.' };
    }
    const missing = flights.filter(f => !f || f.isDeleted === true);
    if (missing.length > 0) {
      return { success: false, error: 'Beberapa data tidak ditemukan atau sudah dihapus.' };
    }

    // ── Pastikan semua berstatus INVOICED ──
    const notInvoiced = flights.filter(f => f.statusFlow !== 'INVOICED');
    if (notInvoiced.length > 0) {
      return {
        success: false,
        error: 'Semua data harus berstatus INVOICED. Data berikut tidak memenuhi syarat: ' +
          notInvoiced.map(f => f.acid + ' (' + f.statusFlow + ')').join(', ')
      };
    }

    // ── Pastikan semua dari unit yang sama ──
    const units = [...new Set(flights.map(f => String(f.unitCode || '').toUpperCase()))];
    if (units.length > 1) {
      return { success: false, error: 'Semua data dalam satu batch harus dari unit yang sama.' };
    }
    const unitCode = units[0];

    // ── Guard: tidak boleh ada rowId yang sedang terikat batch aktif lain ──
    // ✅ OPTIMASI: baca sheet Doc_Batch SEKALI (Map), bukan _getBatchById() per id.
    const refBatchIds = [...new Set(
      flights.map(f => f.batchId ? String(f.batchId).trim() : '').filter(Boolean)
    )];
    const batchByRef = {};
    if (refBatchIds.length) {
      const bs = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.DOC_BATCH);
      if (bs && bs.getLastRow() >= 2) {
        const bh   = bs.getRange(1, 1, 1, bs.getLastColumn()).getValues()[0];
        const bbi  = bh.indexOf('batchId');
        const bsi  = bh.indexOf('status');
        const brows= bs.getRange(2, 1, bs.getLastRow() - 1, bs.getLastColumn()).getValues();
        for (let bi = 0; bi < brows.length; bi++) {
          const bid = String(brows[bi][bbi] || '').trim();
          if (!bid || !refBatchIds.includes(bid)) continue;
          batchByRef[bid] = { batchId: bid, status: String(brows[bi][bsi] || '') };
        }
      }
    }
    const alreadyBatched = flights.filter(f => {
      if (!f.batchId || !String(f.batchId).trim()) return false;
      const existingBatch = batchByRef[String(f.batchId).trim()];
      return existingBatch && ['SENT', 'PROOF_SUBMITTED', 'REJECTED'].includes(existingBatch.status);
    });

    if (alreadyBatched.length > 0) {
      // ── RE-USE: jika SEMUA flight yang diminta berada dalam SATU batch aktif
      // yang sama, jangan buat batch baru — kembalikan batch & link yang sudah ada
      // agar admin cukup kirim ulang via WhatsApp (tidak perlu regenerate). ──
      const batchedIds = [...new Set(alreadyBatched.map(f => String(f.batchId).trim()))];
      const coverCount = batchedIds.reduce((sum, bid) => {
        const covered = flights.filter(f => String(f.batchId || '').trim() === bid).length;
        return sum + covered;
      }, 0);
      if (batchedIds.length === 1 && coverCount === flights.length) {
        const existingBid = batchedIds[0];
        const eb = batchByRef[existingBid];
        const ebFull = _getBatchById(existingBid);
        const baseUrl = CONFIG.PUBLIC_BASE_URL || ScriptApp.getService().getUrl();
        const portalUrl = baseUrl + '/batch.html?bid=' + encodeURIComponent(existingBid) + '&t=' + encodeURIComponent(ebFull ? ebFull.token : '');
        logAudit(ctx.username, 'BATCH_RESEND', 'Doc_Batch', existingBid, null, null, null, null,
          JSON.stringify({ rowIds, reused: true, status: eb ? eb.status : '' }));
        return {
          success: true,
          reused: true,
          batchId: existingBid,
          portalUrl: portalUrl,
          grandTotal: flights.reduce((s, f) => s + (parseFloat(f.totalAmount) || 0), 0),
          count: flights.length,
          operatorName: ebFull && ebFull.operatorName ? ebFull.operatorName : (flights[0].acid || '').substring(0,3),
          airlineCode: ebFull && ebFull.airlineCode ? ebFull.airlineCode : (flights[0].acid || '').substring(0,3),
          expiresAt: (ebFull && ebFull.expiresAt instanceof Date ? ebFull.expiresAt : new Date(ebFull ? ebFull.expiresAt : 0)).toISOString()
        };
      }

      return {
        success: false,
        error: 'Beberapa data sudah terikat batch aktif berbeda: ' +
          alreadyBatched.map(f => f.acid).join(', ') +
          '. Pisahkan data yang sudah punya link, atau pilih hanya satu batch yang sama.'
      };
    }

    // ── Hitung grand total ──
    const grandTotal = flights.reduce((s, f) => s + (parseFloat(f.totalAmount) || 0), 0);

    // ── Ambil info operator dari flight pertama (untuk tampilan portal) ──
    // ✅ FIX: registrasi diawali "PK" (domestik) → gunakan prefix 3-letter
    // kode maskapai; selain itu (callsign non-PK/asing/militer) → gunakan
    // CALLSIGN UTUH. Sebelumnya selalu dipotong 3 huruf, sehingga untuk
    // registrasi non-PK lookup ke Master_Airline gagal cocok (mis. "B7792"
    // terpotong jadi "B77") dan operator jatuh ke fallback potongan itu
    // sendiri, bukan nilai kolom Operator/GH yang sebenarnya.
    const firstFlight  = flights[0];
    const firstAcid    = String(firstFlight.acid || '').toUpperCase().trim();
    const firstReg     = String(firstFlight.registration || '').toUpperCase().trim();
    const airlineCode  = firstReg.startsWith('PK') ? firstAcid.substring(0, 3) : firstAcid;
    let operatorName   = '';
    try {
      const airlineRes = lookupAirlineByCode(airlineCode);
      if (airlineRes && airlineRes.success && airlineRes.data) {
        operatorName = airlineRes.data.operator || airlineRes.data.airlineName || airlineCode;
      }
    } catch(e) {}
    if (!operatorName) operatorName = airlineCode;

    // ── Buat batchId unik & token akses ──
    const batchId    = 'BATCH-' + Utilities.getUuid().replace(/-/g, '').toUpperCase().substring(0, 12);
    const batchToken = Utilities.getUuid().replace(/-/g, '');
    const now        = new Date();
    const expiresAt  = new Date(now.getTime() + BATCH_TOKEN_VALID_DAYS * 24 * 3600 * 1000);

    // ── Simpan ke Doc_Batch ──
    const batchSheet = SpreadsheetApp.getActiveSpreadsheet()
                         .getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    if (!batchSheet) return { success: false, error: 'Sheet Doc_Batch tidak ditemukan. Jalankan initializeSheets() terlebih dahulu.' };

    const bh = batchSheet.getRange(1, 1, 1, batchSheet.getLastColumn()).getValues()[0];
    const batchRow = bh.map(col => {
      if (col === 'batchId')       return batchId;
      if (col === 'unitCode')      return unitCode;
      if (col === 'rowIds')        return JSON.stringify(rowIds.map(id => parseInt(id, 10)));
      if (col === 'operatorName')  return operatorName;
      if (col === 'airlineCode')   return airlineCode;
      if (col === 'token')         return batchToken;
      if (col === 'status')        return 'SENT';
      if (col === 'grandTotal')    return grandTotal;
      if (col === 'reuploadCount') return 0;
      if (col === 'createdAt')     return now;
      if (col === 'createdBy')     return ctx.username;
      if (col === 'expiresAt')     return expiresAt;
      if (col === 'revoked')       return false;
      return '';
    });
    batchSheet.appendRow(batchRow);
    SpreadsheetApp.flush();

    // ── Tandai setiap flight dengan batchId ──
    const flightSheet = SpreadsheetApp.getActiveSpreadsheet()
                          .getSheetByName(CONFIG.SHEETS.FLIGHTS);
    const fh     = flightSheet.getRange(1, 1, 1, flightSheet.getLastColumn()).getValues()[0];
    const bIdIdx = fh.indexOf('batchId');
    const uaIdx  = fh.indexOf('updatedAt');
    const ubIdx  = fh.indexOf('updatedBy');

    if (bIdIdx === -1) {
      return { success: false, error: 'Kolom batchId belum ada di sheet Flights. Jalankan initializeSheets() terlebih dahulu.' };
    }

    rowIds.forEach(rid => {
      const rowNum = parseInt(rid, 10);
      flightSheet.getRange(rowNum, bIdIdx + 1).setValue(batchId);
      if (uaIdx !== -1) flightSheet.getRange(rowNum, uaIdx + 1).setValue(new Date());
      if (ubIdx !== -1) flightSheet.getRange(rowNum, ubIdx + 1).setValue(ctx.username);
    });
    SpreadsheetApp.flush();

    // ── Bangun URL portal publik ──
    // Format: https://airnav-billing.vercel.app/batch?bid=...&t=...
    // Tidak pakai /b/:id/:token karena Vercel rewrite tidak selalu jalan.
    const baseUrl = CONFIG.PUBLIC_BASE_URL || ScriptApp.getService().getUrl();
    const portalUrl = baseUrl + '/batch.html?bid=' + encodeURIComponent(batchId) + '&t=' + encodeURIComponent(batchToken);

    logAudit(ctx.username, 'CREATE_BATCH', 'Doc_Batch', batchId, null, null,
      JSON.stringify({ rowIds, grandTotal, operatorName }));

    return {
      success: true,
      batchId,
      portalUrl,
      grandTotal,
      operatorName,
      airlineCode,
      count: rowIds.length,
      expiresAt: expiresAt.toISOString()
    };

  } catch(e) {
    Logger.log('[generateBatchInvoiceToken] error: ' + e.toString());
    return { success: false, error: e.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Ambil data batch beserta semua flight anggotanya.
 * Dipakai oleh portal publik (Tahap 2) dan panel admin (Tahap 4).
 */
function getBatchData(batchId, batchToken) {
  try {
    if (!batchId || !batchToken) {
      return { success: false, error: 'Parameter tidak lengkap' };
    }

    // Validasi token
    const batch = _getBatchById(batchId);
    if (!batch) return { success: false, error: 'Batch tidak ditemukan' };
    if (batch.revoked === true) return { success: false, error: 'Link batch sudah dicabut' };
    if (String(batch.token).trim() !== String(batchToken).trim()) {
      return { success: false, error: 'Token tidak valid' };
    }
    const exp = batch.expiresAt instanceof Date ? batch.expiresAt : new Date(batch.expiresAt);
    // Jika status PROOF_SUBMITTED → perpanjang otomatis (admin butuh waktu verifikasi)
    if (batch.status !== 'PROOF_SUBMITTED' && Date.now() > exp.getTime()) {
      return { success: false, error: 'Link batch sudah kedaluwarsa' };
    }

    // Ambil semua flight anggota
    let rowIds = [];
    try { rowIds = JSON.parse(batch.rowIds || '[]'); } catch(e) {}
    const flights = rowIds
      .map(rid => getFlightByRowId(parseInt(rid, 10)))
      .filter(f => f && f.isDeleted !== true);

    return _deepSanitizeForClient({
      success: true,
      batch: batch,
      flights: flights,
      count: flights.length,
      grandTotal: parseFloat(batch.grandTotal) || 0
    });
  } catch(e) {
    Logger.log('[getBatchData] error: ' + e.toString());
    return { success: false, error: e.toString() };
  }
}

/**
 * Data untuk portal batch publik — TANPA session, hanya token batch.
 * Mengembalikan data batch + semua flight anggota + data pendukung render dokumen.
 * Dipanggil dari BatchInvoicePortal.html via google.script.run.
 */
function getPublicBatchData(batchId, batchToken) {
  Logger.log('[getPublicBatchData] START batchId=' + batchId);
  try {
    if (!batchId || !batchToken) {
      return { success: false, error: 'Parameter tidak lengkap',
        _diag: { batchId: batchId, batchToken: batchToken ? '***' : null } };
    }

    // ── Validasi batch & token ──
    const batch = _getBatchById(batchId);
    if (!batch) {
      // Cek apakah sheet Doc_Batch ada
      const bs = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.DOC_BATCH);
      const sheetExists = !!bs;
      const rowCount = bs ? Math.max(0, bs.getLastRow() - 1) : 0;
      return {
        success: false,
        error: 'Tagihan tidak ditemukan atau link tidak valid',
        _diag: { batchId, sheetExists, rowCount, hint: 'Jalankan initializeBatchSystem() jika sheet belum ada' }
      };
    }
    if (batch.revoked === true) return { success: false, error: 'Link ini sudah dicabut oleh admin' };
    if (String(batch.token || '').trim() !== String(batchToken).trim()) {
      return {
        success: false, error: 'Token tidak valid',
        _diag: { batchId, storedTokenLen: String(batch.token || '').length, providedTokenLen: String(batchToken || '').length }
      };
    }

    // Cek kedaluwarsa — kecuali status PROOF_SUBMITTED (tunggu verifikasi admin)
    const exp = batch.expiresAt instanceof Date ? batch.expiresAt : new Date(batch.expiresAt || 0);
    if (batch.status !== 'PROOF_SUBMITTED' && Date.now() > exp.getTime()) {
      return {
        success: false, error: 'Link tagihan ini sudah kedaluwarsa. Hubungi admin untuk link baru.',
        _diag: { batchId, expiresAt: exp.toISOString(), now: new Date().toISOString() }
      };
    }

    // ── Ambil semua flight anggota ──
    let rowIds = [];
    try { rowIds = JSON.parse(batch.rowIds || '[]'); } catch(e) {}

    const flights = rowIds
      .map(rid => getFlightByRowId(parseInt(rid, 10)))
      .filter(f => f && f.isDeleted !== true);

    if (flights.length === 0) {
      return {
        success: false, error: 'Data penerbangan tidak ditemukan',
        _diag: { batchId, totalRowIds: rowIds.length, foundFlights: 0 }
      };
    }

    const unitCode = String(batch.unitCode || flights[0].unitCode || '').toUpperCase();

    // ── Data pendukung render invoice ──
    let signRes, bankRes, cfgRes, airlineRes, aptRes, logoRes;
    try { signRes = getSignatory(unitCode); }
    catch(e) { signRes = { success: false, data: null }; }

    try { bankRes = getBankAccounts(unitCode); }
    catch(e) { bankRes = { success: false, data: [] }; }

    try { cfgRes = getConfigData(); }
    catch(e) { cfgRes = { success: false, data: {} }; }

    try { airlineRes = { success: true, data: getMasterData(CONFIG.SHEETS.MASTER_AIRLINE) }; }
    catch(e) { airlineRes = { success: true, data: [] }; }

    try { aptRes = { success: true, data: getMasterData(CONFIG.SHEETS.MASTER_AIRPORT_HOURS) }; }
    catch(e) { aptRes = { success: true, data: [] }; }

    try { logoRes = getLogoBase64(); }
    catch(e) { logoRes = { success: false, data: '' }; }

    const UNITS_MAP = {
      'WAWW':'Cabang Kendari','WAFO':'Unit Morowali','WAWP':'Unit Kolaka',
      'WAWB':'Unit Bau-Bau','WAWR':'Unit Sugimanuru','WAWD':'Unit Wakatobi'
    };

    const rawResult = {
      success:    true,
      batch:      batch,
      flights:    flights,
      grandTotal: parseFloat(batch.grandTotal) || 0,
      unitName:   UNITS_MAP[unitCode] || unitCode,
      signRes,
      bankRes,
      cfgRes,
      airlineRes,
      aptRes,
      logo: logoRes,
    };

    const result = _deepSanitizeForClient(rawResult);

    try { JSON.stringify(result); }
    catch(serErr) {
      Logger.log('[getPublicBatchData] SERIALIZATION ERROR: ' + serErr.toString());
      return { success: false, error: 'Gagal menyusun respons: ' + serErr.toString() };
    }

    Logger.log('[getPublicBatchData] SUCCESS — ' + flights.length + ' flights, status=' + batch.status);
    return result;

  } catch(e) {
    Logger.log('[getPublicBatchData] EXCEPTION: ' + e.toString());
    return { success: false, error: e.toString() };
  }
}

/**
 * Helper internal: ambil (atau buat) folder Drive khusus penyimpanan
 * bukti pembayaran batch. Dipanggil oleh submitBatchProof().
 */
function _getOrCreateBatchProofFolder() {
  const FOLDER_NAME = 'AirNav_BatchPaymentProofs';
  const folders = DriveApp.getFoldersByName(FOLDER_NAME);
  if (folders.hasNext()) return folders.next();
  const folder = DriveApp.createFolder(FOLDER_NAME);
  try {
    folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch(e) {
    Logger.log('[_getOrCreateBatchProofFolder] setSharing gagal: ' + e.toString());
  }
  return folder;
}

/**
 * Terima upload bukti pembayaran dari operator (publik, tanpa login).
 * Dipanggil dari BatchInvoicePortal.html saat operator klik
 * "Kirim Bukti Pembayaran".
 *
 * Validasi:
 *   1. Token & batchId valid, status batch SENT atau REJECTED.
 *   2. File maks 5MB, tipe JPG/PNG/PDF (cek magic bytes).
 *   3. reuploadCount dibatasi (anti-spam, maks 5x).
 *
 * Efek:
 *   - Simpan file ke Drive (folder khusus, sharing anyone-with-link).
 *   - Update Doc_Batch: status=PROOF_SUBMITTED, proofFileUrl/Id, proofNote, proofUploadedAt.
 *   - Update semua Flights anggota: statusFlow INVOICED -> UNPAID (atomik, dengan guard).
 *
 * @param {string} batchId
 * @param {string} token        - token akses publik batch (dari URL portal)
 * @param {string} fileBase64   - data URI base64, mis. "data:image/png;base64,...."
 * @param {string} fileName     - nama file asli (opsional, untuk penamaan Drive)
 * @param {string} mimeType     - mime type file dari client (fallback ke deteksi header)
 * @param {string} note         - catatan/no. referensi transfer (opsional)
 */
function submitBatchProof(batchId, token, fileBase64, fileName, mimeType, note) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return { success: false, error: 'Sistem sedang memproses permintaan lain, coba lagi.' };
  }
  try {
    if (!batchId || !token) {
      return { success: false, error: 'Parameter tidak lengkap' };
    }
    if (!fileBase64 || typeof fileBase64 !== 'string') {
      return { success: false, error: 'Bukti pembayaran wajib diunggah' };
    }

    // ── Validasi batch & token ──
    const batch = _getBatchById(batchId);
    if (!batch) return { success: false, error: 'Batch tidak ditemukan' };
    if (batch.revoked === true) return { success: false, error: 'Link batch sudah dicabut' };
    if (String(batch.token || '').trim() !== String(token).trim()) {
      return { success: false, error: 'Token tidak valid' };
    }
    if (['SENT', 'PROOF_SUBMITTED', 'REJECTED'].indexOf(batch.status) === -1) {
      return {
        success: false,
        error: batch.status === 'VERIFIED'
          ? 'Pembayaran sudah terverifikasi — bukti tidak dapat diganti.'
          : 'Batch ini tidak dapat menerima bukti pembayaran saat ini (status: ' + batch.status + ')'
      };
    }

    // ── Anti-spam re-upload ──
    const reuploadCount = parseInt(batch.reuploadCount, 10) || 0;
    const MAX_REUPLOAD = 5;
    if (reuploadCount >= MAX_REUPLOAD) {
      return { success: false, error: 'Batas maksimal pengiriman ulang bukti tercapai. Hubungi admin.' };
    }

    // ── Jika mengganti bukti lama (status PROOF_SUBMITTED), trash dulu file lama ──
    if (batch.status === 'PROOF_SUBMITTED' && batch.proofFileId) {
      try {
        DriveApp.getFileById(String(batch.proofFileId)).setTrashed(true);
        Logger.log('[submitBatchProof] File bukti lama di-trash: ' + batch.proofFileId);
      } catch(e) {
        Logger.log('[submitBatchProof] Gagal trash file lama (non-fatal): ' + e.toString());
      }
    }

    // ── Parse & validasi file base64 ──
    const match = String(fileBase64).match(/^data:([^;]+);base64,(.+)$/);
    if (!match) return { success: false, error: 'Format file tidak valid' };
    const detectedMime = match[1];
    const base64Data   = match[2];

    const ALLOWED_MIME = ['image/jpeg', 'image/jpg', 'image/png', 'application/pdf'];
    const finalMime = mimeType || detectedMime;
    if (ALLOWED_MIME.indexOf(finalMime) === -1) {
      return { success: false, error: 'Tipe file tidak didukung. Gunakan JPG, PNG, atau PDF.' };
    }

    let bytes;
    try {
      bytes = Utilities.base64Decode(base64Data);
    } catch(e) {
      return { success: false, error: 'Gagal membaca data file' };
    }

    const MAX_SIZE_BYTES = 5 * 1024 * 1024; // 5MB
    if (bytes.length > MAX_SIZE_BYTES) {
      return { success: false, error: 'Ukuran file maksimal 5MB' };
    }
    if (bytes.length < 100) {
      return { success: false, error: 'File terlalu kecil, kemungkinan rusak' };
    }

    // ── Validasi magic bytes (isi file harus sesuai tipe yang diklaim) ──
    const isPNG = bytes[0] === -119 || bytes[0] === 137;
    const isJPG = (bytes[0] & 0xff) === 0xff && (bytes[1] & 0xff) === 0xd8;
    const isPDF = bytes[0] === 0x25 && bytes[1] === 0x50 && bytes[2] === 0x44 && bytes[3] === 0x46; // "%PDF"
    if (!isPNG && !isJPG && !isPDF) {
      return { success: false, error: 'Isi file tidak sesuai dengan tipe yang diklaim (PNG/JPG/PDF)' };
    }

    // ── Simpan ke Drive ──
    const folder   = _getOrCreateBatchProofFolder();
    const safeName = 'BUKTI_' + batchId + '_' + Date.now() +
      (fileName ? ('_' + String(fileName).replace(/[^a-zA-Z0-9._-]/g, '_').substring(0, 60)) : '');
    const blob = Utilities.newBlob(bytes, finalMime, safeName);
    const file = folder.createFile(blob);
    try {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch(e) {
      Logger.log('[submitBatchProof] setSharing file gagal: ' + e.toString());
    }
    const fileUrl = file.getUrl();
    const fileId  = file.getId();

    // ── Update Doc_Batch ──
    const batchSheet = SpreadsheetApp.getActiveSpreadsheet()
                          .getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    const bh   = batchSheet.getRange(1, 1, 1, batchSheet.getLastColumn()).getValues()[0];
    const brow = batch.rowId;
    const wasRejected = batch.status === 'REJECTED';
    const setCell = function(colName, val) {
      const idx = bh.indexOf(colName);
      if (idx !== -1) batchSheet.getRange(brow, idx + 1).setValue(val);
    };
    setCell('status',          'PROOF_SUBMITTED');
    setCell('proofFileUrl',    fileUrl);
    setCell('proofFileId',     fileId);
    setCell('proofNote',       note ? String(note).trim().substring(0, 500) : '');
    setCell('proofUploadedAt', new Date());
    setCell('reuploadCount',   reuploadCount + 1);
    if (wasRejected) {
      // Bersihkan jejak penolakan lama dari sheet (riwayat tetap di Audit_Log)
      setCell('rejectedBy', '');
      setCell('rejectedAt', '');
      setCell('rejectReason', '');
    }
    SpreadsheetApp.flush();

    // ── Update semua Flights anggota: INVOICED -> UNPAID (atomik) ──
    let rowIds = [];
    try { rowIds = JSON.parse(batch.rowIds || '[]'); } catch(e) {}

    const flightSheet = SpreadsheetApp.getActiveSpreadsheet()
                          .getSheetByName(CONFIG.SHEETS.FLIGHTS);
    const fh    = flightSheet.getRange(1, 1, 1, flightSheet.getLastColumn()).getValues()[0];
    const sfIdx = fh.indexOf('statusFlow');
    const rdIdx = fh.indexOf('receiptDate');
    const uaIdx = fh.indexOf('updatedAt');

    let updatedCount = 0;
    rowIds.forEach(function(rid) {
      const rowNum = parseInt(rid, 10);
      if (!rowNum || rowNum < 2) return;
      const currentStatus = flightSheet.getRange(rowNum, sfIdx + 1).getValue();
      // Guard: hanya ubah jika masih INVOICED (mencegah race/duplikasi)
      if (String(currentStatus) !== 'INVOICED') return;
      flightSheet.getRange(rowNum, sfIdx + 1).setValue('UNPAID');
      if (rdIdx !== -1) flightSheet.getRange(rowNum, rdIdx + 1).setValue(new Date().toISOString());
      if (uaIdx !== -1) flightSheet.getRange(rowNum, uaIdx + 1).setValue(new Date());
      updatedCount++;
    });
    SpreadsheetApp.flush();

    logAudit('operator-portal', 'BATCH_PROOF_SUBMITTED', CONFIG.SHEETS.DOC_BATCH, batchId,
      null, batch.status, 'PROOF_SUBMITTED', null,
      JSON.stringify({ batchId: batchId, fileUrl: fileUrl, flightsUpdated: updatedCount, note: note || '' }));

    return {
      success: true,
      status: 'PROOF_SUBMITTED',
      fileUrl: fileUrl,
      flightsUpdated: updatedCount,
      message: 'Bukti pembayaran berhasil dikirim. Menunggu verifikasi admin.'
    };

  } catch(e) {
    Logger.log('[submitBatchProof] error: ' + e.toString());
    return { success: false, error: e.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Hapus bukti pembayaran yang sudah terlanjur diunggah (publik, tanpa login).
 * Dipanggil dari portal BatchInvoicePortal.html saat operator klik "Hapus".
 *
 * Aturan:
 *   - Hanya berlaku saat status batch masih SENT / PROOF_SUBMITTED / REJECTED
 *     (belum diverifikasi). Setelah VERIFIED, bukti tidak bisa dihapus.
 *   - Menghapus file dari Drive (paling aman) dan membersihkan kolom bukti,
 *     lalu status batch dikembalikan ke SENT agar operator bisa unggah baru.
 *   - Flights anggota yang sempat diubah INVOICED->UNPAID DIKEMBALIKAN ke
 *     INVOICED agar konsisten (lihat submitBatchProof yang menaikkan status).
 */
function deleteBatchProof(batchId, token) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return { success: false, error: 'Sistem sedang memproses permintaan lain, coba lagi.' };
  }
  try {
    if (!batchId || !token) {
      return { success: false, error: 'Parameter tidak lengkap' };
    }
    const batch = _getBatchById(batchId);
    if (!batch) return { success: false, error: 'Batch tidak ditemukan' };
    if (batch.revoked === true) return { success: false, error: 'Link batch sudah dicabut' };
    if (String(batch.token || '').trim() !== String(token).trim()) {
      return { success: false, error: 'Token tidak valid' };
    }

    // Bukti hanya bisa dihapus sebelum diverifikasi
    if (batch.status === 'VERIFIED') {
      return { success: false, error: 'Pembayaran sudah terverifikasi — bukti tidak dapat dihapus.' };
    }

    const hasProof = batch.proofFileId || batch.proofFileUrl;
    if (!hasProof) {
      return { success: false, error: 'Tidak ada bukti pembayaran untuk dihapus.' };
    }

    const batchSheet = SpreadsheetApp.getActiveSpreadsheet()
                          .getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    const bh = batchSheet.getRange(1, 1, 1, batchSheet.getLastColumn()).getValues()[0];
    const setCell = function(colName, val) {
      const idx = bh.indexOf(colName);
      if (idx !== -1) batchSheet.getRange(batch.rowId, idx + 1).setValue(val);
    };

    // ── Hapus file dari Drive (jika ID tersedia) ──
    let deletedFromDrive = false;
    if (batch.proofFileId) {

        deletedFromDrive = true;
      } catch(e) {
        Logger.log('[deleteBatchProof] Gagal hapus file Drive: ' + e.toString());
      }
    }

    // ── Bersihkan kolom bukti di Doc_Batch ──
    setCell('status',          'SENT');
    setCell('proofFileUrl',    '');
    setCell('proofFileId',     '');
    setCell('proofNote',       '');
    setCell('proofUploadedAt', '');
    setCell('rejectReason',    '');
    setCell('rejectedBy',      '');

    // ── Revert Flights anggota: UNPAID -> INVOICED ──

                          .getSheetByName(CONFIG.SHEETS.FLIGHTS);
    const fh    = flightSheet.getRange(1, 1, 1, flightSheet.getLastColumn()).getValues()[0];
    const sfIdx = fh.indexOf('statusFlow');
    const rdIdx = fh.indexOf('receiptDate');
    const uaIdx = fh.indexOf('updatedAt');

    let revertedCount = 0;
    rowIds.forEach(function(rid) {
      const rowNum = parseInt(rid, 10);
      if (!rowNum || rowNum < 2) return;
      const currentStatus = flightSheet.getRange(rowNum, sfIdx + 1).getValue();
      if (String(currentStatus) !== 'UNPAID') return;
      flightSheet.getRange(rowNum, sfIdx + 1).setValue('INVOICED');
      if (rdIdx !== -1) flightSheet.getRange(rowNum, rdIdx + 1).setValue('');
      if (uaIdx !== -1) flightSheet.getRange(rowNum, uaIdx + 1).setValue(new Date());
      revertedCount++;
    });
    SpreadsheetApp.flush();

    logAudit('operator-portal', 'BATCH_PROOF_DELETED', CONFIG.SHEETS.DOC_BATCH, batchId,
      null, batch.status, 'SENT', null,
      JSON.stringify({ batchId: batchId, driveDeleted: deletedFromDrive, flightsReverted: revertedCount }));

    return {
      success: true,
      status: 'SENT',
      driveDeleted: deletedFromDrive,
      flightsReverted: revertedCount,
      message: 'Bukti pembayaran berhasil dihapus. Silakan unggah ulang.'
    };

  } catch(e) {
    Logger.log('[deleteBatchProof] error: ' + e.toString());
    return { success: false, error: e.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Ambil detail beberapa batch sekaligus berdasarkan batchId — dipakai
 * admin untuk merender kartu batch terkelompok di tab Pending Kwitansi.
 */
function getBatchesByIds(batchIds, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired', data: [] };
    if (!Array.isArray(batchIds) || batchIds.length === 0) {
      return { success: true, data: [] };
    }
    const uniqueIds = [...new Set(batchIds.map(id => String(id).trim()).filter(Boolean))];
    // ✅ OPTIMASI: baca sheet Doc_Batch SEKALI menjadi Map, lalu cari dari
    // Map — bukan _getBatchById() per id (yang membaca seluruh sheet N kali).
    const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    const rows = [];
    if (s && s.getLastRow() >= 2) {
      const baseUrl = CONFIG.PUBLIC_BASE_URL || ScriptApp.getService().getUrl();
      const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
      const bi   = h.indexOf('batchId');
      const data = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
      const byId = {};
      for (let i = 0; i < data.length; i++) {
        const bid = String(data[i][bi] || '').trim();
        if (!bid) continue;
        if (!byId[bid]) {
          const obj = { rowId: i + 2 };
          h.forEach((k, j) => { obj[k] = data[i][j]; });
          byId[bid] = obj;
        }
      }
      uniqueIds.forEach(id => {
        if (byId[id]) {
          // ── SUMBER TUNGGAL LINK: bangun portalUrl ringkas via PUBLIC_BASE_URL
          // (Vercel), IDENTIK dengan yang dipakai WhatsApp. Client hanya memakai
          // portalUrl ini, tidak rebuild ulang. ──
          const b = byId[id];
          if (b.revoked !== true && b.token) {
            b.portalUrl = baseUrl + '/batch.html?bid=' + encodeURIComponent(String(b.batchId)) +
                          '&t=' + encodeURIComponent(String(b.token));
          }
          rows.push(b);
        }
      });
    }
    return _deepSanitizeForClient({ success: true, data: rows });
  } catch(e) {
    Logger.log('[getBatchesByIds] error: ' + e.toString());
    return { success: false, error: e.toString(), data: [] };
  }
}

/**
 * Admin menolak bukti pembayaran batch — flights kembali ke INVOICED,
 * batch berstatus REJECTED (operator bisa upload ulang lewat portal).
 */
function rejectBatchInvoice(batchId, reasonObj, username, token) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return { success: false, error: 'Sistem sedang memproses permintaan lain, coba lagi.' };
  }
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };
    if (!ctx.isAdmin) {
      logAudit(username, 'UNAUTHORIZED_BATCH_REJECT', 'Doc_Batch', batchId);
      return { success: false, error: 'Hanya admin yang dapat menolak bukti pembayaran' };
    }

    // ── Normalisasi alasan: dukung object {category, detail} atau string lama ──
    let category = '';
    let detail   = '';
    if (reasonObj && typeof reasonObj === 'object') {
      category = String(reasonObj.category || '').trim();
      detail   = String(reasonObj.detail   || '').trim();
    } else {
      detail = String(reasonObj || '').trim();
    }

    // Wajib ada detail minimal 5 karakter (keterangan konkret untuk operator)
    if (!detail || detail.length < 5) {
      return { success: false, error: 'Keterangan penolakan minimal 5 karakter' };
    }
    // Batasi panjang
    detail = detail.substring(0, 500);
    category = category.substring(0, 80);

    if (!category) category = 'Lainnya';

    const batch = _getBatchById(batchId);
    if (!batch) return { success: false, error: 'Batch tidak ditemukan' };
    if (batch.status !== 'PROOF_SUBMITTED') {
      return { success: false, error: 'Batch ini tidak dalam status menunggu verifikasi' };
    }

    const batchSheet = SpreadsheetApp.getActiveSpreadsheet()
                          .getSheetByName(CONFIG.SHEETS.DOC_BATCH);
    const bh = batchSheet.getRange(1, 1, 1, batchSheet.getLastColumn()).getValues()[0];
    const setCell = (col, val) => {
      const idx = bh.indexOf(col);
      if (idx !== -1) batchSheet.getRange(batch.rowId, idx + 1).setValue(val);
    };
    setCell('status', 'REJECTED');
    setCell('rejectedBy', ctx.username || '');
    setCell('rejectedAt', new Date());
    setCell('rejectReason', detail);
    setCell('rejectCategory', category);
    SpreadsheetApp.flush();

    // ── Revert semua Flights anggota: UNPAID -> INVOICED (atomik) ──
    let rowIds = [];
    try { rowIds = JSON.parse(batch.rowIds || '[]'); } catch(e) {}

    const flightSheet = SpreadsheetApp.getActiveSpreadsheet()
                          .getSheetByName(CONFIG.SHEETS.FLIGHTS);
    const fh    = flightSheet.getRange(1, 1, 1, flightSheet.getLastColumn()).getValues()[0];
    const sfIdx = fh.indexOf('statusFlow');
    const rdIdx = fh.indexOf('receiptDate');
    const uaIdx = fh.indexOf('updatedAt');
    const ubIdx = fh.indexOf('updatedBy');

    let updatedCount = 0;
    rowIds.forEach(rid => {
      const rowNum = parseInt(rid, 10);
      if (!rowNum || rowNum < 2) return;
      const currentStatus = flightSheet.getRange(rowNum, sfIdx + 1).getValue();
      if (String(currentStatus) !== 'UNPAID') return;
      flightSheet.getRange(rowNum, sfIdx + 1).setValue('INVOICED');
      if (rdIdx !== -1) flightSheet.getRange(rowNum, rdIdx + 1).setValue('');
      if (uaIdx !== -1) flightSheet.getRange(rowNum, uaIdx + 1).setValue(new Date());
      if (ubIdx !== -1) flightSheet.getRange(rowNum, ubIdx + 1).setValue(ctx.username);
      updatedCount++;
    });
    SpreadsheetApp.flush();

    logAudit(ctx.username, 'BATCH_REJECTED', 'Doc_Batch', batchId, null,
      'PROOF_SUBMITTED', 'REJECTED', null,
      JSON.stringify({ batchId: batchId, category: category, reason: detail }));

    return {
      success: true,
      flightsReverted: updatedCount,
      batchId: batchId,
      category: category,
      reason: detail
    };
  } catch(e) {
    Logger.log('[rejectBatchInvoice] error: ' + e.toString());
    return { success: false, error: e.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Validasi token — dipakai internal oleh getPublicDocData.
 */
function _validateShareToken(rowId, shareToken) {
  if (!rowId || !shareToken) {
    Logger.log('[_validateShareToken] rowId atau shareToken kosong: rid=' + rowId + ' token=' + shareToken);
    return false;
  }
  const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.DOC_SHARE_TOKENS);
  if (!s || s.getLastRow() < 2) {
    Logger.log('[_validateShareToken] Sheet Doc_ShareTokens kosong/tidak ditemukan');
    return false;
  }

  const h = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
  const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
  const ti = h.indexOf('token'), ri = h.indexOf('rowId');
  const ei = h.indexOf('expiresAt'), rvi = h.indexOf('revoked');

  let tokenFound = false;
  for (const row of rows) {
    if (String(row[ti]) !== String(shareToken)) continue;
    tokenFound = true;
    if (String(row[ri]) !== String(rowId)) {
      Logger.log('[_validateShareToken] Token cocok tapi rowId beda: sheet=' + row[ri] + ' url=' + rowId);
      return false;
    }
    if (row[rvi] === true) {
      Logger.log('[_validateShareToken] Token sudah direvoke');
      return false;
    }
    const exp = row[ei] instanceof Date ? row[ei] : new Date(row[ei]);
    if (Date.now() > exp.getTime()) {
      Logger.log('[_validateShareToken] Token kedaluwarsa: exp=' + exp);
      return false;
    }
    return true;
  }
  if (!tokenFound) Logger.log('[_validateShareToken] Token tidak ditemukan di sheet sama sekali: ' + shareToken);
  return false;
}

/**
 * Data untuk halaman download publik — TANPA session, hanya token.
 * Hanya mengembalikan field yang benar-benar dibutuhkan untuk render dokumen.
 */
function getPublicDocData(rowId, shareToken) {
  Logger.log('[getPublicDocData] START rowId=' + rowId + ' (type:' + typeof rowId + ') token=' + shareToken);
  try {
    // ✅ FIX: normalisasi rowId ke Number di titik masuk — rowId dari link
    // publik (google.script.url.getLocation().parameter.rid) selalu string.
    const numericRowId = parseInt(rowId, 10);
    if (!numericRowId || isNaN(numericRowId)) {
      Logger.log('[getPublicDocData] rowId tidak valid setelah parseInt: ' + numericRowId);
      return { success: false, error: 'Link tidak valid atau sudah kedaluwarsa (rowId invalid)' };
    }

    if (!_validateShareToken(numericRowId, shareToken)) {
      Logger.log('[getPublicDocData] Token gagal divalidasi — lihat log _validateShareToken di atas');
      return { success: false, error: 'Link tidak valid atau sudah kedaluwarsa' };
    }
    Logger.log('[getPublicDocData] Token valid, mengambil flight...');

    const flight = getFlightByRowId(numericRowId);
    if (!flight || flight.isDeleted === true) {
      Logger.log('[getPublicDocData] Flight tidak ditemukan / sudah dihapus');
      return { success: false, error: 'Data tidak ditemukan' };
    }
    if (!['INVOICED', 'UNPAID', 'PAID'].includes(flight.statusFlow)) {
      Logger.log('[getPublicDocData] statusFlow belum siap: ' + flight.statusFlow);
      return { success: false, error: 'Dokumen belum tersedia' };
    }
    Logger.log('[getPublicDocData] Flight OK, mengambil data pendukung...');

    // ✅ FIX: setiap pemanggilan sub-fungsi dibungkus try/catch TERPISAH
    // agar SATU sub-fungsi yang gagal (mis. getLogoBase64 karena masalah
    // otorisasi DriveApp untuk pengunjung anonim) tidak menggagalkan
    // seluruh respons — dan supaya Logger.log menunjukkan persis fungsi
    // mana yang bermasalah.
    let signRes, bankRes, cfgRes, airlineData, aptData, logoRes, unitName;
    try { signRes = getSignatory(flight.unitCode); }
    catch(e1) { Logger.log('[getPublicDocData] getSignatory error: ' + e1); signRes = { success:false, data:null }; }

    try { bankRes = getBankAccounts(flight.unitCode); }
    catch(e2) { Logger.log('[getPublicDocData] getBankAccounts error: ' + e2); bankRes = { success:false, data:[] }; }

    try { cfgRes = getConfigData(); }
    catch(e3) { Logger.log('[getPublicDocData] getConfigData error: ' + e3); cfgRes = { success:false, data:{} }; }

    try { airlineData = getMasterData(CONFIG.SHEETS.MASTER_AIRLINE); }
    catch(e4) { Logger.log('[getPublicDocData] getMasterData(Airline) error: ' + e4); airlineData = []; }

    try { aptData = getMasterData(CONFIG.SHEETS.MASTER_AIRPORT_HOURS); }
    catch(e5) { Logger.log('[getPublicDocData] getMasterData(AirportHours) error: ' + e5); aptData = []; }

    try { logoRes = getLogoBase64(); }
    catch(e6) { Logger.log('[getPublicDocData] getLogoBase64 error: ' + e6); logoRes = { success:false, data:'' }; }

    const UNITS_MAP = {'WAWW':'Cabang Kendari','WAFO':'Unit Morowali','WAWP':'Unit Kolaka',
                        'WAWB':'Unit Bau-Bau','WAWR':'Unit Sugimanuru','WAWD':'Unit Wakatobi'};
    unitName = UNITS_MAP[flight.unitCode] || flight.unitCode;

    Logger.log('[getPublicDocData] SUCCESS — mengembalikan hasil lengkap');
    const rawResult = {
      success: true,
      _diag: 'getPublicDocData_v3', // ✅ penanda versi — v3 = sudah pakai deep sanitize
      flight: flight,
      signRes: signRes,
      bankRes: bankRes,
      cfgRes: cfgRes,
      airlineRes: { success: true, data: airlineData },
      aptRes: { success: true, data: aptData },
      logo: logoRes,
      unitName: unitName,
    };

    // ✅ FIX UTAMA: bersihkan SEMUA Date mentah yang bersarang (paling sering
    // ditemukan di signRes.data.updatedAt dari Master_Signatory) sebelum
    // melewati batas RPC google.script.run. Ini yang sebelumnya membuat
    // seluruh respons diterima sebagai null di client, bukan error spesifik.
    const result = _deepSanitizeForClient(rawResult);

    try {
      JSON.stringify(result); // safety-net tambahan untuk kasus lain (fungsi, dsb)
    } catch (serErr) {
      Logger.log('[getPublicDocData] SERIALIZATION ERROR: ' + serErr.toString());
      return { success: false, error: 'Gagal menyusun respons (serialisasi): ' + serErr.toString() };
    }
    return result;
  } catch(e) {
    Logger.log('[getPublicDocData] EXCEPTION TAK TERTANGKAP: ' + e.toString() + '\n' + (e.stack || ''));
    return { success: false, error: e.toString(), _diag: 'getPublicDocData_v3' };
  }
}

function getAppBootstrap(clientVersion, unitCode, username, token) {
  const ctx = getCurrentUserContext(username, token);
  if (!ctx) return { success: false, error: 'Session expired' };
  return {
    success:   true,
    version:   checkAppVersion(clientVersion),
    dashboard: getDashboardData(username, token),
    config:    getConfigData(),
    airlines:  getMasterDataList(CONFIG.SHEETS.MASTER_AIRLINE, username, token),
    schemas:   getMasterSchemas(),

  };
}

function getDashboardData(username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };

    // ✅ PERUBAHAN KEBIJAKAN: dashboard kini transparan untuk SEMUA unit,
    // termasuk non-admin — agar setiap unit bisa melihat performa cluster
    // secara keseluruhan (mendorong produktifitas antar unit).
    // Akses ke DATA DETAIL per baris (getUnitData) TETAP dibatasi ke unit
    // sendiri — hanya AGREGAT dashboard yang dibuka di sini.
    let fl = getAllFlights();

    const units = ["WAWW","WAFO","WAWP","WAWB","WAWR","WAWD"];

    // ✅ OPTIMASI DASHBOARD (1-pass):
    // Sebelumnya: 6× fl.filter(u) lalu 4-5 filter per unit + filter KPI = puluhan
    // pass penuh atas array. Kini SATU loop tunggal menghitung per-unit counter,
    // KPI total, distribusi kategori/dom, dan byYear sekaligus.

    // Per-unit counters
    const uc = {};
    units.forEach(u => { uc[u] = { validation:0, invoice:0, kwitansi:0, paid:0, nominal:0, revenue:0 }; });

    // KPI total
    let kpiTotal     = 0,
        kpiInvoice   = 0,
        kpiPaid      = 0,
        kpiPending   = 0,
        kpiNominal   = 0,
        kpiRevenue   = 0,
        extCount     = 0,
        advCount     = 0,
        domCount     = 0,
        intCount     = 0;
    const byYear = {};

    for (let idx = 0; idx < fl.length; idx++) {
      const r  = fl[idx];
      const u  = String(r.unitCode || '');
      const st = String(r.statusFlow || '');
      const amt = parseFloat(r.totalAmount) || 0;
      const isPaid = st === 'PAID';

      // Per-unit
      if (uc[u]) {
        const c = uc[u];
        if (st === 'VALIDATION') c.validation++;
        else if (st === 'INVOICED') c.invoice++;
        else if (st === 'UNPAID') c.kwitansi++;
        else if (isPaid) { c.paid++; c.revenue += amt; }
        c.nominal += amt;
      }

      // KPI total
      kpiTotal++;
      if (st === 'INVOICED') kpiInvoice++;
      else if (st === 'PAID') { kpiPaid++; kpiRevenue += amt; }
      else if (st === 'UNPAID') kpiPending++;
      kpiNominal += amt;

      // Distribusi kategori & domInt (normalisasi)
      const cat = String(r.category || '').toUpperCase().trim();
      const dom = String(r.domInt || '').toUpperCase().trim();
      const validCat = (cat === 'EXTEND' || cat === 'ADVANCE') ? cat : 'UNKNOWN';
      const validDom = (dom === 'DOMESTIK' || dom === 'INTERNATIONAL') ? dom : 'UNKNOWN';
      if (validCat === 'EXTEND') extCount++;
      if (validCat === 'ADVANCE') advCount++;
      if (validDom === 'DOMESTIK') domCount++;
      if (validDom === 'INTERNATIONAL') intCount++;

      // byYear
      const y = r.flightDate ? String(r.flightDate).substring(0, 4) : null;
      if (y && y.length === 4) {
        if (!byYear[y]) byYear[y] = { flights: [], ext: 0, adv: 0, dom: 0, int: 0 };
        const yr = byYear[y];
        yr.flights.push({
          unitCode: r.unitCode,
          statusFlow: st,
          totalAmount: amt,
          category: validCat,
          domInt: validDom,
          acid: String(r.acid || '').toUpperCase().trim(),
          flightDate: r.flightDate || '',
          invoiceNo: r.invoiceNo || '',
          rowId: r.rowId || (idx + 2),
          validatedAt: r.validatedAt || '',
          receiptDate: r.receiptDate || '',
        });
        if (validCat === 'EXTEND') yr.ext++;
        if (validCat === 'ADVANCE') yr.adv++;
        if (validDom === 'DOMESTIK') yr.dom++;
        if (validDom === 'INTERNATIONAL') yr.int++;
      }
    }

    const badges = {};
    const ur = [];
    // ── Bangun badges & ur dari uc ──
    units.forEach(u => {
        const c = uc[u];
        badges[u] = { validation: c.validation, invoice: c.invoice, kwitansi: c.kwitansi };
        ur.push({ unit: u, validation: c.validation, invoice: c.invoice, kwitansi: c.kwitansi, paid: c.paid, nominal: c.nominal, revenue: c.revenue });
    });

    return {
      success: true,
      data: {
        kpi: {
          totalData:    kpiTotal,
          totalInvoice: kpiInvoice,
          totalPaid:    kpiPaid,
          totalPending: kpiPending,
          totalNominal: kpiNominal,
          totalRevenue: kpiRevenue,
          unitResumes: ur,
          extCount, advCount, domCount, intCount,
        },
        badges,
        byYear,
        availableYears: Object.keys(byYear).map(Number).sort((a,b) => b - a),
      }
    };
  } catch(e) {
    return { success: false, error: e.toString() };
  }
}

function getUnitData(unitCode, username, token) {
  try {
    if (!unitCode) return { success: false, error: 'unitCode kosong', data: [] };

    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired — silakan login ulang', data: [] };
    if (!ctx.isAdmin && String(unitCode).toUpperCase() !== ctx.unitCode) {
      logAudit(username, 'UNAUTHORIZED_UNIT_DATA_ACCESS', 'Flights', null,
        null, null, null, null, `Tried to access ${unitCode} as ${ctx.unitCode}`);
      return { success: false, error: 'Tidak memiliki akses ke unit ini', data: [] };
    }

    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.FLIGHTS);
    if (!s || s.getLastRow() < 2) return { success: true, data: [] };

    const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
    const ui   = h.indexOf('unitCode');
    const di   = h.indexOf('isDeleted');
    const data = [];

    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      if (String(row[ui]) !== String(unitCode)) continue;
      const d = row[di];
      if (d === true || String(d).toUpperCase() === 'TRUE') continue;
      const o = { rowId: i + 2 };
      h.forEach((k, j) => {
        let v = row[j];
        if (v instanceof Date) {
          if (k === 'flightDate') {
            v = v.getFullYear() + '-' +
                String(v.getMonth() + 1).padStart(2,'0') + '-' +
                String(v.getDate()).padStart(2,'0');
          } else if (k === 'atd' || k === 'ata') {
            v = String(v.getHours()).padStart(2,'0') + ':' +
                String(v.getMinutes()).padStart(2,'0');
          } else {
            v = v.toISOString();
          }
        }
        o[k] = (v === null || v === undefined) ? '' : v;
      });
      data.push(o);
    }
    JSON.stringify(data); // validasi serialisasi
    return { success: true, data };
  } catch(e) {
    return { success: false, error: e.toString(), data: [] };
  }
}

function saveNewFlight(data, user, token) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000); 
    const ctx = getCurrentUserContext(user, token); 
    if (!ctx) return { success: false, error: 'Session expired' };
    
    if (!data || typeof data !== 'object')   
      return { success: false, error: 'Data tidak valid' };
    
    // ✅ PERBAIKAN: Gunakan 'data'
    const validation = validateFlightInput(data);
    if (!validation.valid) {
      return { success: false, error: validation.errors.join('; ') };
    }
    
    // Non-admin hanya boleh input untuk unit sendiri
    if (!ctx.isAdmin && String(data.unitCode).toUpperCase() !== ctx.unitCode) {
      logAudit(ctx.username, 'UNAUTHORIZED_CREATE_ATTEMPT', 'Flights', null);
      return { success: false, error: 'Tidak dapat input data untuk unit lain' };
    }
    
    // Sanitasi semua string input
    Object.keys(data).forEach(key => {
      if (typeof data[key] === 'string') {
        data[key] = data[key].trim().substring(0, 200); 
      }
    });
    
    const u = ctx.username;
    if (!data.flightDate || !/^\d{4}-\d{2}-\d{2}$/.test(String(data.flightDate)))
      return { success: false, error: 'Format tanggal tidak valid' };
    if (!data.acid || String(data.acid).trim() === '')
      return { success: false, error: 'ACID wajib diisi' };
    if (!data.registration || String(data.registration).trim() === '')
      return { success: false, error: 'Registrasi wajib diisi' };
    if (!data.unitCode || !['WAWW','WAFO','WAWP','WAWB','WAWR','WAWD'].includes(data.unitCode))
      return { success: false, error: 'Unit code tidak valid' };

    // ── Idempotency: cegah duplikat dalam 15 detik ──
    const fingerprint = [
      data.unitCode,
      data.flightDate,
      String(data.acid).toUpperCase().trim(),
      String(data.registration).toUpperCase().trim(),
      data.dep_arr_loc || ''
    ].join('|');

    const cache    = CacheService.getScriptCache();
    // ✅ GANTI DENGAN — hash SHA256 untuk collision-free key:
    // Hash fingerprint agar key pendek, unik, dan tidak bisa di-manipulasi
    const fpBytes = Utilities.computeDigest(
      Utilities.DigestAlgorithm.SHA_256,
      fingerprint
    );
    const fpHash = fpBytes.map(b => ('0' + (b & 0xff).toString(16)).slice(-2)).join('').substring(0, 40);
    const cacheKey = 'fp_' + fpHash;
    
    if (cache.get(cacheKey)) {
      return {
        success: false,
        error: 'Data identik baru saja disimpan. Tunggu beberapa detik sebelum mencoba lagi.',
        duplicate: true
      };
    }
    // Perpanjang window menjadi 30 detik untuk mencegah double-click di koneksi lambat
    cache.put(cacheKey, '1', 30);

    data.statusFlow = CONFIG.STATUS.VALIDATION;
    delete data.grossAmount; delete data.vatAmount; delete data.pphAmount;
    delete data.totalAmount; delete data.invoiceNo; delete data.validatedAt;
    delete data.validatedBy; delete data.signatorySnapshot;

    return saveFlight(data, u); // ✅ Sudah benar menggunakan 'data'

  } catch (e) {
    return { success: false, error: "Gagal menyimpan: " + e.message }; // Jangan throw, return object agar frontend bisa baca
  } finally {
    lock.releaseLock(); 
  }
}

function updateFlightData(rid, updates, username, token) {
  if (!username) username = Session.getActiveUser().getEmail();
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired — silakan login ulang' };
    
    const flight = getFlightByRowId(rid);
    if (!flight) return { success: false, error: 'Data tidak ditemukan' };
    
    const access = canAccessFlight(username, token, flight);
    if (!access.allowed) {
      return { success: false, error: access.reason };
    }

    // ── Validasi setiap field yang diubah — sebelumnya hanya saveNewFlight yang
    //    tervalidasi, sehingga jalur EDIT bisa dipakai menyisipkan teks bebas
    //    (termasuk markup HTML) ke field yang seharusnya enum tetap. ──
    const fieldErrors = [];
    if ('flightDate' in updates && !VALIDATORS.flightDate(updates.flightDate)) fieldErrors.push('Tanggal tidak valid');
    if ('acid' in updates && !VALIDATORS.acid(updates.acid)) fieldErrors.push('ACID tidak valid');
    if ('registration' in updates && !VALIDATORS.registration(updates.registration)) fieldErrors.push('Registrasi tidak valid');
    if ('aircraftType' in updates && updates.aircraftType && !VALIDATORS.aircraftType(updates.aircraftType)) fieldErrors.push('Tipe pesawat tidak valid');
    if ('adep' in updates && updates.adep && !VALIDATORS.airportCode(updates.adep)) fieldErrors.push('ADEP tidak valid');
    if ('ades' in updates && updates.ades && !VALIDATORS.airportCode(updates.ades)) fieldErrors.push('ADES tidak valid');
    if ('statusFlight' in updates && !['REGULER','CHARTER','FERYFLIGHT','MILITER','KALIBRASI'].includes(String(updates.statusFlight).toUpperCase())) fieldErrors.push('Status flight tidak valid');
    if ('category' in updates && !['EXTEND','ADVANCE'].includes(String(updates.category).toUpperCase())) fieldErrors.push('Kategori tidak valid');
    if ('domInt' in updates && !['DOMESTIK','INTERNATIONAL'].includes(String(updates.domInt).toUpperCase())) fieldErrors.push('Dom/Int tidak valid');
    if ('dep_arr_loc' in updates && !['DEP','ARR','LOC'].includes(String(updates.dep_arr_loc).toUpperCase())) fieldErrors.push('DEP/ARR/LOC tidak valid');
    if (fieldErrors.length) return { success: false, error: fieldErrors.join('; ') };
    
    // Non-admin tidak boleh mengubah field-field kritis
    if (!ctx.isAdmin) {
      const FORBIDDEN_FIELDS = [
        'statusFlow', 'invoiceNo', 'grossAmount', 'vatAmount', 
        'pphAmount', 'totalAmount', 'validatedAt', 'validatedBy'
      ];
      FORBIDDEN_FIELDS.forEach(field => {
        if (field in updates) delete updates[field];
      });
      
      // Non-admin tidak boleh edit data yang sudah INVOICED
      if (['INVOICED','UNPAID','PAID','VOID'].includes(String(flight.statusFlow))) {
        return { success: false, error: 'Data sudah diproses — tidak dapat diubah' };
      }
    }
    
  return updateFlight(rid, updates, ctx.username, 'UPDATE');
  } catch(e) { 
    return { success: false, error: e.toString() }; 
  }
}

// Whitelist sheet yang boleh diakses lewat endpoint generik ini — mencegah
// pemanggilan dengan nama sheet arbitrer (mis. 'Flights' atau 'Audit_Log')
// yang seharusnya hanya melalui jalur khusus dengan filter/otorisasi sendiri.
const MASTER_LIST_ALLOWED_SHEETS = new Set([
  CONFIG.SHEETS.MASTER_AIRLINE, CONFIG.SHEETS.MASTER_RATE, CONFIG.SHEETS.MASTER_USER,
  CONFIG.SHEETS.MASTER_CONFIG, CONFIG.SHEETS.MASTER_BANK, CONFIG.SHEETS.MASTER_SIGNATORY,
  CONFIG.SHEETS.MASTER_AIRPORT_HOURS, CONFIG.SHEETS.MASTER_EXCHANGE_RATE,
  CONFIG.SHEETS.MASTER_SEQUENCE, CONFIG.SHEETS.INVOICE_SETTINGS
]);

function getMasterDataList(sheet, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };

    if (!MASTER_LIST_ALLOWED_SHEETS.has(sheet)) {
      logAudit(username, 'INVALID_MASTER_SHEET_ACCESS', String(sheet), null);
      return { success: false, error: 'Sheet tidak diizinkan' };
    }

    // Master_User berisi hash password — akses dibatasi admin, dan hash
    // TIDAK PERNAH dikirim mentah ke client, sekalipun ke admin.
    if (sheet === CONFIG.SHEETS.MASTER_USER && !ctx.isAdmin) {
      logAudit(username, 'UNAUTHORIZED_USER_LIST_ACCESS', sheet, null);
      return { success: false, error: 'Akses ditolak' };
    }

    const data = getMasterData(sheet).map(row => {
      const o = {};
      Object.keys(row).forEach(k => {
        if (sheet === CONFIG.SHEETS.MASTER_USER && k === 'passwordHash') {
          o[k] = row[k] ? '••••••••' : ''; // placeholder — hash asli tidak pernah keluar server
          return;
        }
        o[k] = sv(row[k]);
      });
      return o;
    });
    return { success: true, data };
  } catch(e) { return { success: false, error: e.toString() }; }
}

function saveMasterDataItem(sheet, data, rid, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired — silakan login ulang' };

    if (!MASTER_LIST_ALLOWED_SHEETS.has(sheet)) {
      logAudit(username, 'INVALID_MASTER_SHEET_WRITE', String(sheet), rid);
      return { success: false, error: 'Sheet tidak diizinkan untuk diedit lewat endpoint ini' };
    }
    
    // Master data HANYA boleh diubah oleh ADMIN
    if (!ctx.isAdmin) {
      logAudit(username, 'UNAUTHORIZED_MASTER_EDIT', sheet, rid);
      return { success: false, error: 'Hanya administrator yang dapat mengubah master data' };
    }
    
    // Validasi khusus untuk Master_User — password harus di-hash
    if (sheet === CONFIG.SHEETS.MASTER_USER && data.passwordHash) {
      const pw = String(data.passwordHash).trim();
      if (pw === '••••••••') {
        // Placeholder tidak diubah admin → jangan sentuh password yang sudah ada
        delete data.passwordHash;
      } else if (pw && !pw.startsWith('pbkdf2$')) {
        // Validasi password policy
        const policy = validatePasswordPolicy(pw, data.username || '');
        if (!policy.valid) {
          return { success: false, error: 'Password: ' + policy.reason };
        }
        // Hash password baru
        try {
          data.passwordHash = hashPassword(pw);
        } catch (e) {
          return { success: false, error: 'Gagal meng-hash password: ' + e.message };
        }
      }
    }
    
    // Proteksi field sensitif di Master_Config
    if (sheet === CONFIG.SHEETS.MASTER_CONFIG) {
      const PROTECTED_KEYS = ['company_name', 'company_address'];
      if (PROTECTED_KEYS.includes(data.configKey) && rid) {
        // Izinkan edit tapi log sebagai perubahan kritis
        logAudit(username, 'CRITICAL_CONFIG_CHANGE', sheet, rid);
      }
    }
    
    const result = saveMasterData(sheet, data, ctx.username, rid);

    // ✅ AUTO-QR SIGNATURE: QR digenerate OTOMATIS dari NIP setiap kali
    // signatory disimpan (tambah/ubah). Admin tidak perlu menginput teks QR
    // manual — cukup isi NIP. Kolom qrcodeUrl (URL lama) sudah dihapus.
    if (result.success && sheet === CONFIG.SHEETS.MASTER_SIGNATORY) {
      const nip = String(data.nip || '').trim();
      const targetRid = rid || result.rowId;
      if (nip && targetRid && targetRid > 1) {
        try {
          generateAndSaveQRSnapshot(nip, targetRid, username, token);
        } catch(qrErr) {
          Logger.log('[saveMasterDataItem] auto-QR signature gagal (non-fatal): ' + qrErr);
        }
      }
    }

    return result;
  } catch(e) { 
    return { success: false, error: e.toString() }; 
  }
}

function deleteMasterDataItem(sheet, rid, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx) return { success: false, error: 'Session expired' };
    
    // Delete master data HANYA untuk admin
    if (!ctx.isAdmin) {
      logAudit(username, 'UNAUTHORIZED_MASTER_DELETE', sheet, rid);
      return { success: false, error: 'Hanya administrator yang dapat menghapus master data' };
    }
    
    // Proteksi: jangan hapus data kritis
    const PROTECTED_SHEETS = [CONFIG.SHEETS.MASTER_CONFIG];
    if (PROTECTED_SHEETS.includes(sheet)) {
      // Untuk config, cek apakah key yang dihapus adalah critical
      const existing = getMasterData(sheet).find(r => r.rowId === rid);
      if (existing) {
        const CRITICAL_KEYS = ['vat_percent', 'pph_percent', 'default_kurs_usd', 'company_name'];
        if (CRITICAL_KEYS.includes(existing.configKey)) {
          logAudit(username, 'CRITICAL_CONFIG_DELETE_ATTEMPT', sheet, rid);
          return { success: false, error: `Config "${existing.configKey}" adalah data kritis — tidak dapat dihapus. Edit saja nilainya.` };
        }
      }
    }
    
    // Proteksi: jangan hapus admin terakhir
    if (sheet === CONFIG.SHEETS.MASTER_USER) {
      const users = getMasterData(sheet).filter(u => u.isActive !== false);
      if (users.length <= 1) {
        return { success: false, error: 'Tidak dapat menghapus — harus ada minimal 1 user aktif' };
      }
    }
    
    return softDeleteMasterData(sheet, rid, ctx.username);
  } catch(e) { 
    return { success: false, error: e.toString() }; 
  }
}

/**
 * Lookup data airline berdasarkan 3-letter code (prefix ACID).
 * Dipanggil dari modal validasi di frontend.
 */
function lookupAirlineByCode(code) {
  try {
    if (!code) return { success: false, data: null };
    const upper = String(code).toUpperCase().trim();
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.MASTER_AIRLINE);
    if (!s || s.getLastRow() < 2) return { success: true, data: null };
    const h    = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
    const ci   = h.indexOf('airlineCode');
    const oi   = h.indexOf('operator');
    const ni   = h.indexOf('airlineName');
    const wi   = h.indexOf('waNumber');
    const ei   = h.indexOf('email');
    const di   = h.indexOf('isDeleted');
    for (let i = 0; i < rows.length; i++) {
      if (String(rows[i][ci]).toUpperCase().trim() !== upper) continue;
      if (di !== -1 && (rows[i][di] === true || String(rows[i][di]).toUpperCase() === 'TRUE')) continue;
      return {
        success: true,
        data: {
          rowId:       i + 2,
          airlineCode: rows[i][ci] || '',
          operator:    oi !== -1 ? (rows[i][oi] || '') : '',
          airlineName: ni !== -1 ? (rows[i][ni] || '') : '',
          waNumber:    wi !== -1 ? (rows[i][wi] || '') : '',
          email:       ei !== -1 ? (rows[i][ei] || '') : '',
        }
      };
    }
    return { success: true, data: null }; // tidak ditemukan
  } catch(e) {
    return { success: false, error: e.toString(), data: null };
  }
}

/**
 * Simpan atau update data airline dari modal validasi.
 * Jika rowId ada → update, jika tidak → insert baru.
 */
function saveAirlineFromValidation(airlineCode, operator, airlineName, rowId, username) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    if (!airlineCode) return { success: false, error: 'Kode maskapai wajib diisi' };
    const upper = String(airlineCode).toUpperCase().trim();
    const s = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName(CONFIG.SHEETS.MASTER_AIRLINE);
    if (!s) return { success: false, error: 'Sheet Master_Airline tidak ditemukan' };
    const h  = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];
    const ci = h.indexOf('airlineCode');
    const oi = h.indexOf('operator');
    const ni = h.indexOf('airlineName');
    const di = h.indexOf('isDeleted');
    const uid = (username && String(username).trim()) ? String(username).trim()
    : (Session.getActiveUser().getEmail() || 'system');

    if (rowId && rowId > 1) {
      // UPDATE baris yang sudah ada
      const existRow = s.getRange(rowId, 1, 1, h.length).getValues()[0];
      if (oi !== -1 && operator)    existRow[oi] = String(operator).toUpperCase().trim();
      if (ni !== -1 && airlineName) existRow[ni] = String(airlineName).toUpperCase().trim();
      s.getRange(rowId, 1, 1, h.length).setValues([existRow]);
      logAudit(uid, 'UPDATE', CONFIG.SHEETS.MASTER_AIRLINE, rowId);
      return { success: true, action: 'updated', rowId };
    } else {
      // Cek duplikat terlebih dahulu
      if (s.getLastRow() >= 2) {
        const rows = s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getValues();
        for (let i = 0; i < rows.length; i++) {
          if (String(rows[i][ci]).toUpperCase().trim() === upper &&
              (di === -1 || rows[i][di] !== true)) {
            // Sudah ada — update saja
            const existRowId = i + 2;
            const existRow = rows[i];
            if (oi !== -1 && operator)    existRow[oi] = String(operator).toUpperCase().trim();
            if (ni !== -1 && airlineName) existRow[ni] = String(airlineName).toUpperCase().trim();
            s.getRange(existRowId, 1, 1, h.length).setValues([existRow]);
            logAudit(uid, 'UPDATE', CONFIG.SHEETS.MASTER_AIRLINE, existRowId);
            return { success: true, action: 'updated', rowId: existRowId };
          }
        }
      }
      // INSERT baru
      const newRow = h.map(col => {
        if (col === 'airlineCode') return upper;
        if (col === 'operator')    return operator ? String(operator).toUpperCase().trim() : '';
        if (col === 'airlineName') return airlineName ? String(airlineName).toUpperCase().trim() : '';
        if (col === 'isDeleted')   return false;
        return '';
      });
      s.appendRow(newRow);
      SpreadsheetApp.flush();
      const newRowId = s.getLastRow();
      logAudit(uid, 'CREATE', CONFIG.SHEETS.MASTER_AIRLINE, newRowId);
      return { success: true, action: 'created', rowId: newRowId };
    }
  } finally {
    lock.releaseLock();
  }
}

function getSignatory(uc) {
  try {
    const s = getMasterData(CONFIG.SHEETS.MASTER_SIGNATORY);
    const norm = function(v){ return String(v || '').trim().toUpperCase(); };
    const target = norm(uc);

    // 1) Cari signatory aktif dengan unitCode yang cocok (case/space-insensitive)
    let sign = s.find(x => norm(x.unitCode) === target && x.isActive !== false);
    // 2) Fallback ke baris ALL aktif (signatory pusat)
    if (!sign) sign = s.find(x => norm(x.unitCode) === "ALL" && x.isActive !== false);
    // 3) Fallback ke signatory pertama yang punya data QR (base64) — pastikan QR selalu tampil
    if (!sign) {
      sign = s.find(x => x.qrcodeBase64 && String(x.qrcodeBase64).indexOf('data:image') === 0);
    }
    // 4) Fallback terakhir: baris aktif pertama apa pun
    if (!sign) sign = s.find(x => x.isActive !== false);

    return { success: true, data: sign || null };
  } catch(e) { return { success: false, error: e.toString() }; }
}

// ==================== NOTIFIKASI SISTEM ====================
function getNotifications(unitCode, lastCheckedISO, callerUsername, token) {
  try {
    // ✅ SECURITY: unitCode & username WAJIB dari session tervalidasi,
    // bukan dari parameter client — sebelumnya bisa dipalsukan.
    const ctx = getCurrentUserContext(callerUsername, token);
    if (!ctx) return { success: false, data: [] };
    unitCode       = ctx.unitCode;
    callerUsername = ctx.username;

    const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.AUDIT_LOG);
    const lastRow = s.getLastRow();
    if (!s || lastRow < 2) return { success: true, data: [] };

    // ✅ FIX: Hanya baca 300 baris terakhir untuk menghemat kuota & memori
    const startRow = Math.max(2, lastRow - 300);
    const rows = s.getRange(startRow, 1, lastRow - startRow + 1, s.getLastColumn()).getValues();

    // ✅ TAMBAHKAN BARIS INI
    const h = s.getRange(1, 1, 1, s.getLastColumn()).getValues()[0];

    const ti  = h.indexOf('timestamp');
    const ui  = h.indexOf('userId');
    const ai  = h.indexOf('action');
    const tbi = h.indexOf('tableName');
    const ri  = h.indexOf('recordId');
    const nvi = h.indexOf('newValue');

    // Fallback 10 menit ke belakang jika lastChecked tidak dikirim
    const since = lastCheckedISO
      ? new Date(lastCheckedISO)
      : new Date(Date.now() - 10 * 60 * 1000);
    
    // Validasi: jika since lebih dari 24 jam lalu, batasi ke 1 jam (cegah flood)
    const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);
    const effectiveSince = since < oneHourAgo ? oneHourAgo : since;

    // callerUsername = username yang login di client (dikirim eksplisit)
    // JANGAN pakai Session.getActiveUser() — tidak reliable di Web App mode "Execute as: Me"
    const currentUser = (callerUsername || '').toLowerCase().trim();

    // ✅ OPTIMASI NOTIFIKASI:
    // Sebelumnya kita baca SELURUH sheet Flights di SETIAP poll (tiap 15 detik
    // per user) hanya untuk context. Kini kita kumpulkan recordId flight yang
    // RIIL dirujuk oleh audit di window ini, lalu baca HANYA baris itu (satu
    // getRange menggabungkan rentang rowId terkecil-sampai-terbesar).
    const flightMap = {};
    {
      const needed = [];
      const ri2    = h.indexOf('recordId');
      const ti2    = h.indexOf('timestamp');
      const ai2    = h.indexOf('action');
      const tbi2   = h.indexOf('tableName');
      for (let i = 0; i < rows.length; i++) {
        const tsRow = rows[i][ti2];
        if (!tsRow) continue;
        const dRow  = tsRow instanceof Date ? tsRow : new Date(tsRow);
        if (isNaN(dRow) || dRow <= effectiveSince) continue;
        if (String(rows[i][tbi2]) !== 'Flights') continue;
        const rid = rows[i][ri2];
        if (rid !== null && rid !== undefined && rid !== '') needed.push(parseInt(rid));
      }
      if (needed.length) {
        const uniq = [...new Set(needed.filter(n => !isNaN(n) && n > 0))];
        if (uniq.length) {
          const frSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEETS.FLIGHTS);
          if (frSheet && frSheet.getLastRow() >= 2) {
            const fh2 = frSheet.getRange(1, 1, 1, frSheet.getLastColumn()).getValues()[0];
            const minR = Math.min(...uniq), maxR = Math.max(...uniq);
            const maxRow = Math.min(maxR, frSheet.getLastRow());
            if (maxRow >= minR) {
              const frows2 = frSheet.getRange(minR, 1, maxRow - minR + 1, frSheet.getLastColumn()).getValues();
              const fUi2  = fh2.indexOf('unitCode');
              const fAc2  = fh2.indexOf('acid');
              const fDa2  = fh2.indexOf('flightDate');
              const fSt2  = fh2.indexOf('statusFlow');
              const fIn2  = fh2.indexOf('invoiceNo');
              for (let k = 0; k < frows2.length; k++) {
                const rowId = minR + k;
                if (!uniq.includes(rowId)) continue;
                const fr = frows2[k];
                flightMap[rowId] = {
                  unitCode:  String(fr[fUi2]  || ''),
                  acid:      String(fr[fAc2]  || ''),
                  flightDate:fr[fDa2] instanceof Date ? fmtDate(fr[fDa2]) : String(fr[fDa2] || ''),
                  statusFlow:String(fr[fSt2] || ''),
                  invoiceNo: String(fr[fIn2] || ''),
                };
              }
            }
          }
        }
      }
    }

    const notifs = [];

    for (let i = rows.length - 1; i >= 0; i--) {
      const row = rows[i];
      const ts  = row[ti];
      if (!ts) continue;
      const tsDate = ts instanceof Date ? ts : new Date(ts);
      if (isNaN(tsDate) || tsDate <= effectiveSince) continue;

      const action = String(row[ai]  || '');
      const actor  = String(row[ui]  || '');
      const table  = String(row[tbi] || '');
      const recId  = row[ri];
      const newVal = String(row[nvi] || '');

      // Jangan kirim ke aktor sendiri (sudah dapat toast lokal)
      // Bandingkan dengan username (bukan email) yang dikirim dari client
      const actorUsername = actor.split('@')[0].toLowerCase();
      if (currentUser && (actor.toLowerCase() === currentUser ||
                          actorUsername === currentUser)) continue;

      const flight = flightMap[recId];
      // ── GATING UNIT & ROLE (PERBAIKAN HAK AKSES NOTIFIKASI) 
      const isCallerAdmin = String(unitCode || '').toUpperCase() === 'ALL';
        
        if (!isCallerAdmin) {
            // NON-ADMIN: HANYA boleh menerima notifikasi untuk alur data awal:
            // Input (CREATE), Edit (UPDATE), Hapus (DELETE), dan Validasi (VALIDATE).
            // Status billing (UNPAID, PAID), VOID, Login, dll TIDAK dikirim ke non-admin.
            const allowedActionsForStaff = ['CREATE', 'UPDATE', 'DELETE', 'VALIDATE'];
            if (!allowedActionsForStaff.includes(action)) continue;
            
            // Pastikan hanya notifikasi untuk unitnya sendiri yang masuk
            const flightUnit = flight ? String(flight.unitCode || '').toUpperCase() : null;
            if (!flightUnit || flightUnit !== String(unitCode).toUpperCase()) continue;
        }

      let type    = 'info';
      let message = '';
      let icon    = 'fa-bell';
      let sound   = 'chime'; // chime | alert | success | warning

      if (action === 'CREATE' && table === 'Flights') {
        type    = 'info';
        icon    = 'fa-plane';
        sound   = 'chime';
        const f = flight || {};
        message = `Data baru masuk: ${f.acid || 'pesawat'} — ${f.flightDate || ''} (${actor.split('@')[0]})`;

      } else if (action === 'VALIDATE') {
        type    = 'success';
        icon    = 'fa-check-circle';
        sound   = 'success';
        const f = flight || {};
        message = `Invoice ${f.invoiceNo || newVal || ''} digenerate — ${f.acid || '—'} (${actor.split('@')[0]})`;

      } else if (action === 'STATUS_CHANGE') {
        const statusLabel = { UNPAID: 'Pending Kwitansi', PAID: 'Lunas ✓' };
        const f = flight || {};
        type  = newVal === 'PAID' ? 'success' : 'info';
        icon  = newVal === 'PAID' ? 'fa-check-double' : 'fa-exchange-alt';
        sound = newVal === 'PAID' ? 'success' : 'chime';
        message = `Status ${f.acid || '—'} berubah → ${statusLabel[newVal] || newVal} (${actor.split('@')[0]})`;

      } else if (action === 'VOID') {
        type    = 'warn';
        icon    = 'fa-ban';
        sound   = 'warning';
        const f = flight || {};
        message = `Invoice ${f.invoiceNo || '—'} dibatalkan oleh ${actor.split('@')[0]}`;

      } else if (action === 'UPDATE' && table === 'Flights') {
        type    = 'info';
        icon    = 'fa-pen';
        sound   = 'chime';
        const f = flight || {};
        message = `Data ${f.acid || '—'} — ${f.flightDate || ''} diperbarui (${actor.split('@')[0]})`;
      } else if (action === 'DELETE' && table === 'Flights') {
        type    = 'warn';
        icon    = 'fa-trash-alt';
        sound   = 'warning';
        const f = flight || {};
        message = `Data ${f.acid || '—'} dihapus oleh ${actor.split('@')[0]}`;

      } else if (action === 'ARCHIVE') {
        type    = 'info';
        icon    = 'fa-box-archive';
        sound   = 'soft';
        const f = flight || {};
        message = `Invoice ${f.invoiceNo || f.acid || '—'} selesai — masuk Billing History (${actor.split('@')[0]})`;

      } else if (action === 'BATCH_PROOF_SUBMITTED') {
        type    = 'warn';
        icon    = 'fa-file-circle-check';
        sound   = 'warning';
        message = `Bukti pembayaran batch diterima dari operator — menunggu verifikasi admin`;
        // Override 'action' field agar sampai ke client untuk trigger banner
        // (field ini dibaca di fetchNotifications client)

      } else if (action === 'BATCH_VERIFIED') {
        type    = 'success';
        icon    = 'fa-check-double';
        sound   = 'success';
        message = `Pembayaran batch terkonfirmasi lunas — data masuk Billing History`;

      } else if (action === 'BATCH_REJECTED') {
        type    = 'warn';
        icon    = 'fa-times-circle';
        sound   = 'warning';
        message = `Bukti pembayaran batch ditolak — operator perlu upload ulang`;

      } else if (action === 'LOGIN') {
        type    = 'info';
        icon    = 'fa-sign-in-alt';
        sound   = 'chime';
        message = `${actor.split('@')[0]} login ke sistem`;

      } else {
        continue;
      }

      notifs.push({
        id:        `${i}_${tsDate.getTime()}`,
        type,
        icon,
        sound,
        message,
        action,      // ✅ TAMBAHKAN: digunakan client untuk trigger banner khusus
        actor:     actor.split('@')[0],
        timestamp: tsDate.toISOString(),
        read:      false
      });

      if (notifs.length >= 30) break;
    }

    return { success: true, data: notifs.reverse() };

  } catch(e) {
    Logger.log('getNotifications error: ' + e.toString());
    return { success: false, data: [], error: e.toString() };
  }
}

// ==================== INPUT VALIDATION ====================
const VALIDATORS = {
  flightDate: (v) => {
    if (!v || !/^\d{4}-\d{2}-\d{2}$/.test(String(v))) return false;
    const d = new Date(v);
    if (isNaN(d.getTime())) return false;
    const now = new Date();
    const minDate = new Date(now.getFullYear() - 5, 0, 1);
    const maxDate = new Date(now.getFullYear() + 1, 11, 31);
    return d >= minDate && d <= maxDate;
  },
  
  acid: (v) => {
    if (!v || typeof v !== 'string') return false;
    // Normalisasi toleran: buang spasi/tanda hubung/titik yang sering diketik user
    const t = v.trim().toUpperCase().replace(/[\s\-.]/g, '');
    // Realitas operasional: callsign bisa berupa kode maskapai+nomor (GIA123),
    // registrasi tanpa hubung (B77JC, N123AB, 9MABC), atau callsign khusus.
    // Guard hanya menjaga hygiene: alfanumerik murni, 2-8 karakter
    // (FPL Item 7 maks 7 karakter; +1 toleransi).
    return t.length >= 2 && t.length <= 8 && /^[A-Z0-9]+$/.test(t);
  },
  
  registration: (v) => {
    if (!v || typeof v !== 'string') return false;
    const trimmed = v.trim().toUpperCase();
    // PK-ABC format atau alphanumeric 4-8 char
    return /^[A-Z0-9\-]{4,10}$/.test(trimmed);
  },
  
  aircraftType: (v) => {
    if (!v) return true; // optional
    const trimmed = String(v).trim().toUpperCase();
    return /^[A-Z0-9\-\/\s]{2,20}$/.test(trimmed);
  },
  
  airportCode: (v) => {
    if (!v || typeof v !== 'string') return false;
    const trimmed = v.trim().toUpperCase();
    return /^[A-Z]{4}$/.test(trimmed); // ICAO 4-letter code
  },
  
  time: (v) => {
    if (!v || typeof v !== 'string') return false;
    return /^([01]\d|2[0-3]):([0-5]\d)$/.test(v.trim());
  },
  
  unitCode: (v) => {
    const VALID_UNITS = ['WAWW','WAFO','WAWP','WAWB','WAWR','WAWD','ALL'];
    return VALID_UNITS.includes(String(v).toUpperCase().trim());
  },
  
  kurs: (v) => {
    const n = parseFloat(v);
    return !isNaN(n) && n >= 1000 && n <= 100000;
  },
  
  amount: (v) => {
    const n = parseFloat(v);
    return !isNaN(n) && n >= 0 && n <= 1e12; // max 1 triliun
  }
};

/**
 * Validasi input object terhadap schema
 * Dipanggil sebelum save/update flight
 */
function validateFlightInput(data) {
  const errors = [];
  
  if (!VALIDATORS.flightDate(data.flightDate)) 
    errors.push('Tanggal penerbangan tidak valid');
  if (!VALIDATORS.acid(data.acid))
    errors.push('ACID tidak valid (2-8 karakter huruf/angka, tanpa spasi & simbol)');
  if (!VALIDATORS.registration(data.registration)) 
    errors.push('Registrasi tidak valid');
  if (data.aircraftType && !VALIDATORS.aircraftType(data.aircraftType))
    errors.push('Tipe pesawat tidak valid');
  if (data.adep && !VALIDATORS.airportCode(data.adep))
    errors.push('Kode bandara asal (ADEP) tidak valid');
  if (data.ades && !VALIDATORS.airportCode(data.ades))
    errors.push('Kode bandara tujuan (ADES) tidak valid');
  if (!VALIDATORS.unitCode(data.unitCode))
    errors.push('Unit code tidak valid');
  if (data.statusFlight && !['REGULER','CHARTER','FERYFLIGHT','MILITER','KALIBRASI'].includes(String(data.statusFlight).toUpperCase()))
    errors.push('Status flight tidak valid');
  if (data.category && !['EXTEND','ADVANCE'].includes(String(data.category).toUpperCase()))
    errors.push('Kategori tidak valid');
  if (data.domInt && !['DOMESTIK','INTERNATIONAL'].includes(String(data.domInt).toUpperCase()))
    errors.push('Dom/Int tidak valid');
  
  return {
    valid: errors.length === 0,
    errors
  };
}

// ==================== QR CODE GENERATOR & SNAPSHOT ====================

/**
 * Generate QR Code dari teks menggunakan QR Server API,
 * simpan sebagai Base64 di Master_Signatory.
 *
 * Dipanggil dari frontend saat admin input/edit signatory.
 * Hasilnya disimpan ke kolom qrcodeBase64 — tidak perlu fetch ulang saat PDF.
 *
 * @param {string} text     - Teks/URL yang di-encode ke QR
 * @param {number} rowId    - rowId di Master_Signatory
 * @param {string} username - Username admin
 * @param {string} token    - Session token
 */
function generateAndSaveQRSnapshot(text, rowId, username, token) {
  try {
    const ctx = getCurrentUserContext(username, token);
    if (!ctx || !ctx.isAdmin) {
      return { success: false, error: 'Hanya admin yang dapat mengubah data signatory' };
    }
    if (!text || String(text).trim().length === 0) {
      return { success: false, error: 'Teks QR tidak boleh kosong' };
    }
    if (!rowId || rowId < 2) {
      return { success: false, error: 'rowId tidak valid' };
    }

    // Sanitasi teks — cegah injeksi URL
    const safeText = String(text).trim().substring(0, 500);

    // ── Fetch QR dari API publik (tidak butuh key) ──
    // Alternatif: goqr.me (lebih stabil), qrserver.com
    // Format: PNG 200x200, margin 1, error correction L
    const qrUrl = 'https://api.qrserver.com/v1/create-qr-code/?' +
      'size=200x200' +
      '&margin=4' +
      '&ecc=M' +           // error correction Medium — keseimbangan ukuran & ketahanan
      '&format=png' +
      '&data=' + encodeURIComponent(safeText);

    let qrBase64 = '';
    try {
      const response = UrlFetchApp.fetch(qrUrl, {
        muteHttpExceptions: true,
        headers: { 'User-Agent': 'AirNav-Billing/1.0' }
      });

      if (response.getResponseCode() !== 200) {
        return {
          success: false,
          error: 'Gagal fetch QR dari server (HTTP ' + response.getResponseCode() + ')'
        };
      }

      const blob  = response.getBlob();
      const bytes = blob.getBytes();

      // Validasi: pastikan response adalah PNG (magic bytes: 89 50 4E 47)
      if (bytes[0] !== -119 && bytes[0] !== 137) { // 0x89 = 137 atau -119 (signed)
        return { success: false, error: 'Response bukan gambar PNG valid' };
      }

      qrBase64 = 'data:image/png;base64,' + Utilities.base64Encode(bytes);

    } catch(fetchErr) {
      Logger.log('[generateAndSaveQRSnapshot] Fetch error: ' + fetchErr);
      return { success: false, error: 'Gagal menghubungi server QR: ' + fetchErr.message };
    }

    if (!qrBase64 || qrBase64.length < 500) {
      return { success: false, error: 'Data QR terlalu kecil, kemungkinan kosong' };
    }

    // ── Simpan ke sheet Master_Signatory ──
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_SIGNATORY);
    if (!sheet) return { success: false, error: 'Sheet Master_Signatory tidak ditemukan' };
    if (rowId > sheet.getLastRow()) {
      return { success: false, error: 'rowId melebihi jumlah baris sheet' };
    }

    const headers  = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const b64Idx   = headers.indexOf('qrcodeBase64');
    const textIdx  = headers.indexOf('qrcodeText');   // kolom opsional — simpan teks asli
    const uaIdx    = headers.indexOf('updatedAt');

    if (b64Idx === -1) {
      // Kolom belum ada — tambahkan secara otomatis (migrasi)
      const newCol = headers.length + 1;
      sheet.getRange(1, newCol).setValue('qrcodeBase64');
      sheet.getRange(rowId, newCol).setValue(qrBase64);
    } else {
      sheet.getRange(rowId, b64Idx + 1).setValue(qrBase64);
    }

    // Simpan teks QR jika kolom ada
    if (textIdx !== -1) {
      sheet.getRange(rowId, textIdx + 1).setValue(safeText);
    }

    // Update timestamp
    if (uaIdx !== -1) {
      sheet.getRange(rowId, uaIdx + 1).setValue(new Date());
    }

    SpreadsheetApp.flush();

    logAudit(ctx.username, 'QR_SNAPSHOT_SAVED', CONFIG.SHEETS.MASTER_SIGNATORY,
      rowId, 'qrcodeBase64', null, '[base64 data]', null, safeText.substring(0, 80));

    // Kembalikan base64 ke frontend agar bisa langsung preview
    return {
      success:   true,
      base64:    qrBase64,
      textUsed:  safeText,
      sizeKB:    Math.round(qrBase64.length * 0.75 / 1024)
    };

  } catch(e) {
    Logger.log('[generateAndSaveQRSnapshot] Error: ' + e.toString());
    return { success: false, error: e.message };
  }
}

/**
 * Ambil QR Base64 dari snapshot yang tersimpan.
 * Dipanggil saat generate PDF — tidak ada fetch eksternal.
 *
 * @param {string} unitCode - Kode unit (WAWW, WAFO, dll)
 * @returns {string|null} Base64 data URI atau null jika belum ada
 */
function getQRSnapshotByUnit(unitCode) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_SIGNATORY);
    if (!sheet || sheet.getLastRow() < 2) return null;

    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const rows    = sheet.getRange(2, 1, sheet.getLastRow()-1,
                      sheet.getLastColumn()).getValues();

    const ucIdx  = headers.indexOf('unitCode');
    const b64Idx = headers.indexOf('qrcodeBase64');
    const aiIdx  = headers.indexOf('isActive');

    if (ucIdx === -1 || b64Idx === -1) return null;

    // Cari signatory untuk unit ini (prioritas: exact match, fallback: ALL)
    let exactRow = null, fallbackRow = null;

    for (const row of rows) {
      const uc     = String(row[ucIdx] || '').toUpperCase().trim();
      const active = aiIdx === -1 || row[aiIdx] !== false;
      if (!active) continue;

      if (uc === String(unitCode).toUpperCase()) exactRow    = row;
      else if (uc === 'ALL')                     fallbackRow = row;
    }

    const targetRow = exactRow || fallbackRow;
    if (!targetRow) return null;

    const b64 = String(targetRow[b64Idx] || '');
    return b64.startsWith('data:image') ? b64 : null;

  } catch(e) {
    Logger.log('[getQRSnapshotByUnit] Error: ' + e.toString());
    return null;
  }
}

/**
 * Migrasi: fetch & simpan QR untuk semua signatory yang belum punya snapshot.
 * Jalankan sekali dari Apps Script Editor (Run → migrateQRSnapshots).
 */
function migrateQRSnapshots() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEETS.MASTER_SIGNATORY);
  if (!sheet || sheet.getLastRow() < 2) {
    Logger.log('[migrateQRSnapshots] Sheet kosong atau tidak ditemukan');
    return;
  }

  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const rows    = sheet.getRange(2, 1, sheet.getLastRow()-1,
                    sheet.getLastColumn()).getValues();

  const ucIdx   = headers.indexOf('unitCode');
  const urlIdx  = headers.indexOf('qrcodeUrl');
  const b64Idx  = headers.indexOf('qrcodeBase64');
  const nameIdx = headers.indexOf('name');

  // Buat kolom qrcodeBase64 jika belum ada
  let actualB64Idx = b64Idx;
  if (b64Idx === -1) {
    const newCol = headers.length + 1;
    sheet.getRange(1, newCol).setValue('qrcodeBase64');
    actualB64Idx = newCol - 1; // 0-based index
    Logger.log('[migrateQRSnapshots] Kolom qrcodeBase64 dibuat di kolom ' + newCol);
  }

  let migrated = 0, skipped = 0, failed = 0;

  for (let i = 0; i < rows.length; i++) {
    const rowNum = i + 2;
    const name   = nameIdx !== -1 ? String(rows[i][nameIdx] || '') : '';
    const uc     = ucIdx   !== -1 ? String(rows[i][ucIdx]   || '') : '';

    // Skip jika sudah punya snapshot
    const existing = actualB64Idx < rows[i].length
      ? String(rows[i][actualB64Idx] || '') : '';
    if (existing.startsWith('data:image')) {
      Logger.log('[migrate] Skip (sudah ada): ' + name + ' (' + uc + ')');
      skipped++;
      continue;
    }

    // Ambil dari URL jika ada, atau generate dari nama
    const existingUrl = urlIdx !== -1 ? String(rows[i][urlIdx] || '') : '';
    let qrText = existingUrl || name; // fallback: encode nama signatory

    if (!qrText) {
      Logger.log('[migrate] Skip (tidak ada data): baris ' + rowNum);
      skipped++;
      continue;
    }

    try {
      // Jika ada URL → coba fetch gambar langsung
      if (existingUrl && existingUrl.startsWith('http')) {
        try {
          const resp = UrlFetchApp.fetch(existingUrl, { muteHttpExceptions: true });
          if (resp.getResponseCode() === 200) {
            const b64 = 'data:image/png;base64,' +
              Utilities.base64Encode(resp.getBlob().getBytes());
            sheet.getRange(rowNum, actualB64Idx + 1).setValue(b64);
            Logger.log('[migrate] ✅ Fetch URL berhasil: ' + name);
            migrated++;
            Utilities.sleep(300); // jaga rate limit
            continue;
          }
        } catch(fe) {
          Logger.log('[migrate] URL fetch gagal, fallback ke QR generate: ' + fe);
        }
      }

      // Generate QR baru dari teks/nama
      const qrUrl = 'https://api.qrserver.com/v1/create-qr-code/?' +
        'size=200x200&margin=4&ecc=M&format=png' +
        '&data=' + encodeURIComponent(qrText.substring(0, 200));

      const resp = UrlFetchApp.fetch(qrUrl, { muteHttpExceptions: true });
      if (resp.getResponseCode() === 200) {
        const b64 = 'data:image/png;base64,' +
          Utilities.base64Encode(resp.getBlob().getBytes());
        sheet.getRange(rowNum, actualB64Idx + 1).setValue(b64);
        Logger.log('[migrate] ✅ QR generate berhasil: ' + name + ' (' + uc + ')');
        migrated++;
      } else {
        Logger.log('[migrate] ❌ QR generate gagal HTTP ' +
          resp.getResponseCode() + ': ' + name);
        failed++;
      }

    } catch(e) {
      Logger.log('[migrate] ❌ Error baris ' + rowNum + ': ' + e);
      failed++;
    }

    Utilities.sleep(500); // rate limit antar request
  }

  SpreadsheetApp.flush();
  Logger.log('[migrateQRSnapshots] Selesai — migrated: ' + migrated +
    ', skipped: ' + skipped + ', failed: ' + failed);
}

/**
 * Jalankan fungsi ini dari Apps Script Editor untuk memastikan
 * sheet Doc_Batch dan kolom batchId di Flights sudah ada.
 * Cukup jalankan SEKALI, lalu redeploy.
 */
function initializeBatchSystem() {
  Logger.log('[init] Memulai inisialisasi batch system...');

  // 1. Buat sheet Doc_Batch jika belum ada
  createSheetIfNeeded(CONFIG.SHEETS.DOC_BATCH, [
    "batchId","unitCode","rowIds","operatorName","airlineCode",
    "token","status","grandTotal",
    "proofFileUrl","proofFileId","proofNote","proofUploadedAt",
    "verifiedBy","verifiedAt",
    "rejectedBy","rejectedAt","rejectReason","rejectCategory",
    "reuploadCount",
    "expiresAt","revoked"
  ]);
  Logger.log('[init] Sheet Doc_Batch OK');

  // 2. Pastikan kolom batchId ada di sheet Flights
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const flSheet = ss.getSheetByName(CONFIG.SHEETS.FLIGHTS);
  if (flSheet) {
    const lastCol = flSheet.getLastColumn();
    const headers = lastCol > 0
      ? flSheet.getRange(1, 1, 1, lastCol).getValues()[0]
      : [];
    if (headers.indexOf('batchId') === -1) {
      flSheet.getRange(1, headers.length + 1).setValue('batchId');
      Logger.log('[init] Kolom batchId ditambahkan ke sheet Flights');
    } else {
      Logger.log('[init] Kolom batchId sudah ada di Flights');
    }
  }

  // 3. Pastikan sheet Doc_ShareTokens ada
  createSheetIfNeeded(CONFIG.SHEETS.DOC_SHARE_TOKENS,
    ["token","rowId","expiresAt","revoked"]);
  Logger.log('[init] Sheet Doc_ShareTokens OK');

  SpreadsheetApp.flush();
  Logger.log('[init] ✅ Batch system initialization selesai!');
  return { success: true, message: 'Batch system initialized. Silakan redeploy Web App.' };
}

/**
 * Diagnostik untuk memeriksa apakah batch system sudah siap.
 * Panggil dari Apps Script Editor atau via google.script.run.
 */
function diagnoseBatchSystem() {
  const result = { sheets: {}, columns: {}, functions: {} };

  // Cek sheet
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ['DOC_BATCH', 'FLIGHTS', 'DOC_SHARE_TOKENS'].forEach(key => {
    const name = CONFIG.SHEETS[key];
    const sheet = ss.getSheetByName(name);
    result.sheets[key] = sheet ? 'OK (' + sheet.getLastRow() + ' rows)' : 'MISSING';
  });

  // Cek kolom batchId di Flights
  const flSheet = ss.getSheetByName(CONFIG.SHEETS.FLIGHTS);
  if (flSheet && flSheet.getLastColumn() > 0) {
    const headers = flSheet.getRange(1, 1, 1, flSheet.getLastColumn()).getValues()[0];
    result.columns.batchId = headers.indexOf('batchId') !== -1 ? 'OK' : 'MISSING';
  }

  // Cek jumlah batch
  const batchSheet = ss.getSheetByName(CONFIG.SHEETS.DOC_BATCH);
  if (batchSheet && batchSheet.getLastRow() > 1) {
    result.batchCount = batchSheet.getLastRow() - 1;
  } else {
    result.batchCount = 0;
  }

  // Cek fungsi
  result.functions.getPublicBatchData = typeof getPublicBatchData === 'function' ? 'OK' : 'MISSING';
  result.functions.generateBatchInvoiceToken = typeof generateBatchInvoiceToken === 'function' ? 'OK' : 'MISSING';
  result.functions.submitBatchProof = typeof submitBatchProof === 'function' ? 'OK' : 'MISSING';

  Logger.log('[diagnose] ' + JSON.stringify(result, null, 2));
  return result;
}
